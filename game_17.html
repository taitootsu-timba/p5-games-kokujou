// ============================================================
// 3D FPS MAZE
// p5.js
// ============================================================


// ============================================================
// MAZE
// ============================================================

const maze = [
  "###############",
  "#P            #",
  "# ##### ##### #",
  "#     #       #",
  "##### # #######",
  "#     #       #",
  "# ### ##### # #",
  "# #         # #",
  "# # ####### # #",
  "# #       #   #",
  "# ####### #####",
  "#             #",
  "###############"
];


// ============================================================
// SETTINGS
// ============================================================

const CELL_SIZE = 80;
const WALL_HEIGHT = 120;

const PLAYER_HEIGHT = 45;
const PLAYER_RADIUS = 18;

const MOVE_SPEED = 4;

const MOUSE_SENSITIVITY = 0.003;


// ============================================================
// PLAYER
// ============================================================

let playerX = 0;
let playerZ = 0;

let playerAngle = 0;


// ============================================================
// KEYBOARD
// ============================================================

let keys = {
  w: false,
  a: false,
  s: false,
  d: false
};


// ============================================================
// WALLS
// ============================================================

let walls = [];


// ============================================================
// SETUP
// ============================================================

function setup() {

  createCanvas(
    windowWidth,
    windowHeight,
    WEBGL
  );

  createMaze();

  noStroke();

  // ブラウザのキーイベント
  window.addEventListener(
    "keydown",
    handleKeyDown
  );

  window.addEventListener(
    "keyup",
    handleKeyUp
  );

  // ウィンドウから離れたら停止
  window.addEventListener(
    "blur",
    releaseAllKeys
  );

}


// ============================================================
// KEY DOWN
// ============================================================

function handleKeyDown(event) {

  if (event.code === "KeyW") {
    keys.w = true;
  }

  if (event.code === "KeyA") {
    keys.a = true;
  }

  if (event.code === "KeyS") {
    keys.s = true;
  }

  if (event.code === "KeyD") {
    keys.d = true;
  }

}


// ============================================================
// KEY UP
// ============================================================

function handleKeyUp(event) {

  if (event.code === "KeyW") {
    keys.w = false;
  }

  if (event.code === "KeyA") {
    keys.a = false;
  }

  if (event.code === "KeyS") {
    keys.s = false;
  }

  if (event.code === "KeyD") {
    keys.d = false;
  }

}


// ============================================================
// RELEASE ALL KEYS
// ============================================================

function releaseAllKeys() {

  keys.w = false;
  keys.a = false;
  keys.s = false;
  keys.d = false;

}


// ============================================================
// CREATE MAZE
// ============================================================

function createMaze() {

  const mazeWidth =
    maze[0].length;

  const mazeHeight =
    maze.length;


  const offsetX =
    mazeWidth * CELL_SIZE / 2;

  const offsetZ =
    mazeHeight * CELL_SIZE / 2;


  for (
    let row = 0;
    row < mazeHeight;
    row++
  ) {

    for (
      let col = 0;
      col < maze[row].length;
      col++
    ) {

      const tile =
        maze[row][col];


      const x =
        col * CELL_SIZE
        - offsetX
        + CELL_SIZE / 2;


      const z =
        row * CELL_SIZE
        - offsetZ
        + CELL_SIZE / 2;


      if (tile === "#") {

        walls.push({
          x: x,
          z: z
        });

      }


      if (tile === "P") {

        playerX = x;
        playerZ = z;

      }

    }

  }

}


// ============================================================
// DRAW
// ============================================================

function draw() {

  background(
    30,
    35,
    45
  );


  updatePlayer();

  updateCamera();


  // Lighting

  ambientLight(100);

  directionalLight(
    255,
    255,
    255,
    -0.5,
    -1,
    -0.5
  );


  drawFloor();

  drawWalls();

}


// ============================================================
// PLAYER UPDATE
// ============================================================

function updatePlayer() {

  let forward = 0;
  let strafe = 0;


  // ------------------------------
  // Forward
  // ------------------------------

  if (keys.w) {
    forward += 1;
  }

  if (keys.s) {
    forward -= 1;
  }


  // ------------------------------
  // Strafe
  // ------------------------------

  if (keys.a) {
    strafe -= 1;
  }

  if (keys.d) {
    strafe += 1;
  }


  // ------------------------------
  // Stop
  // ------------------------------

  if (
    forward === 0 &&
    strafe === 0
  ) {

    return;

  }


  // ------------------------------
  // Direction
  // ------------------------------

  let dx =
    sin(playerAngle) *
    forward;

  let dz =
    -cos(playerAngle) *
    forward;


  // ------------------------------
  // Strafe
  // ------------------------------

  dx +=
    cos(playerAngle) *
    strafe;

  dz +=
    sin(playerAngle) *
    strafe;


  // ------------------------------
  // Normalize
  // ------------------------------

  const length =
    sqrt(
      dx * dx +
      dz * dz
    );


  if (length === 0) {
    return;
  }


  dx /= length;
  dz /= length;


  // ------------------------------
  // Speed
  // ------------------------------

  dx *= MOVE_SPEED;
  dz *= MOVE_SPEED;


  // ------------------------------
  // Collision
  // ------------------------------

  const nextX =
    playerX + dx;


  if (
    !isColliding(
      nextX,
      playerZ
    )
  ) {

    playerX = nextX;

  }


  const nextZ =
    playerZ + dz;


  if (
    !isColliding(
      playerX,
      nextZ
    )
  ) {

    playerZ = nextZ;

  }

}


// ============================================================
// COLLISION
// ============================================================

function isColliding(
  x,
  z
) {

  for (
    const wall of walls
  ) {

    const closestX =
      constrain(
        x,
        wall.x - CELL_SIZE / 2,
        wall.x + CELL_SIZE / 2
      );


    const closestZ =
      constrain(
        z,
        wall.z - CELL_SIZE / 2,
        wall.z + CELL_SIZE / 2
      );


    const distance =
      dist(
        x,
        z,
        closestX,
        closestZ
      );


    if (
      distance < PLAYER_RADIUS
    ) {

      return true;

    }

  }


  return false;

}


// ============================================================
// CAMERA
// ============================================================

function updateCamera() {

  const lookDistance = 100;


  const lookX =
    playerX +
    sin(playerAngle) *
    lookDistance;


  const lookZ =
    playerZ -
    cos(playerAngle) *
    lookDistance;


  camera(

    playerX,
    -PLAYER_HEIGHT,
    playerZ,

    lookX,
    -PLAYER_HEIGHT,
    lookZ,

    0,
    1,
    0

  );

}


// ============================================================
// FLOOR
// ============================================================

function drawFloor() {

  push();

  translate(
    0,
    20,
    0
  );

  rotateX(
    HALF_PI
  );

  ambientMaterial(
    50,
    55,
    60
  );

  plane(
    3000,
    3000
  );

  pop();

}


// ============================================================
// WALLS
// ============================================================

function drawWalls() {

  for (
    const wall of walls
  ) {

    push();

    translate(
      wall.x,
      -30,
      wall.z
    );

    ambientMaterial(
      120,
      130,
      145
    );

    box(
      CELL_SIZE,
      WALL_HEIGHT,
      CELL_SIZE
    );

    pop();

  }

}


// ============================================================
// MOUSE
// ============================================================

function mousePressed() {

  if (
    !document.pointerLockElement
  ) {

    requestPointerLock();

  }

}


function mouseMoved() {

  if (
    document.pointerLockElement
  ) {

    playerAngle +=
      movedX *
      MOUSE_SENSITIVITY;

  }

}


// ============================================================
// POINTER LOCK CHANGE
// ============================================================

document.addEventListener(
  "pointerlockchange",
  function() {

    // Pointer Lockが解除されたら
    // 移動キーを全部リセット

    if (
      !document.pointerLockElement
    ) {

      releaseAllKeys();

    }

  }
);


// ============================================================
// WINDOW RESIZE
// ============================================================

function windowResized() {

  resizeCanvas(
    windowWidth,
    windowHeight
  );

}
