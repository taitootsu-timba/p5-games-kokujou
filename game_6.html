let magic = {
  damage: 45,

  cooldown: 2000,

  missileSpeed: 7,

  missileCount: 1,

  pierce: 0,

  explosion: 0,

  lightning: 0,

  ice: 0,

  meteor: 0,

  tornado: 0,

  holy: 0,

  heal: 0

// =====================================================
// SETUP
// =====================================================

function setup() {

  createCanvas(
    min(windowWidth, 480),
    windowHeight
  );

  castle = {
    x: width / 2,

    y: height - 75,

    width: 135,

    height: 65,

    hp: 2000,

    maxHp: 2000
  };

  spellTimer = millis();

  specialTimer = millis();

  textFont("sans-serif");
}


// =====================================================
// DRAW
// =====================================================

function draw() {

  drawBackground();

  if (gameState === "play") {

    updateGame();

    drawGame();

  } else if (gameState === "levelup") {

    drawGame();

    drawLevelUp();

  } else if (gameState === "gameover") {

    drawGame();

    drawGameOver();
  }
}


// =====================================================
// GAME UPDATE
// =====================================================

function updateGame() {

  spawnEnemies();

  updateEnemies();

  autoSpell();

  updateMissiles();

  updateSpecialMagic();

  updateEffects();

  updateParticles();

  if (magic.heal > 0) {

    castle.hp = min(
      castle.maxHp,
      castle.hp +
      magic.heal *
      0.015 *
      deltaTime
    );
  }

  if (castle.hp <= 0) {

    castle.hp = 0;

    gameState = "gameover";
  }
}

// =====================================================
// SPAWN
// =====================================================

function spawnEnemies() {

  spawnTimer += deltaTime;


  /*
    最初は約2秒に1体。

    レベルが上がると
    少しずつ早くなる。
  */

  let interval =
    max(
      850,
      2000 -
      (level - 1) * 70
    );


  if (
    spawnTimer <
    interval
  ) {
    return;
  }


  spawnTimer = 0;


  /*
    序盤は絶対に1体。

    Lv5以降から
    低確率で2体。
  */

  let amount = 1;


  if (
    level >= 5 &&
    random() < 0.12
  ) {

    amount = 2;
  }


  if (
    level >= 10 &&
    random() < 0.18
  ) {

    amount = 2;
  }


  if (
    level >= 18 &&
    random() < 0.20
  ) {

    amount = 3;
  }


  for (
    let i = 0;
    i < amount;
    i++
  ) {

    createEnemy();
  }
}


// =====================================================
// CREATE ENEMY
// =====================================================

function createEnemy() {

  let roll =
    random();

  let type;


  if (
    roll < 0.72
  ) {

    type = "normal";

  } else if (
    roll < 0.94
  ) {

    type = "fast";

  } else {

    type = "tank";
  }


  let hp;
  let speed;
  let radius;
  let damage;
  let reward;
  let color;


  // NORMAL

  if (
    type === "normal"
  ) {

    hp =
      38 +
      level * 7;

    speed =
      0.55 +
      level * 0.012;

    radius = 15;

    damage =
      1.0 +
      level * 0.035;

    reward = 1;

    color = [
      185,
      60,
      75
    ];
  }


  // FAST

  else if (
    type === "fast"
  ) {

    hp =
      60 +
      level * 9;

    speed =
      0.82 +
      level * 0.015;

    radius = 12;

    damage =
      1.4 +
      level * 0.04;

    reward = 2;

    color = [
      230,
      110,
      55
    ];
  }


  // TANK

  else {

    hp =
      160 +
      level * 22;

    speed =
      0.32 +
      level * 0.007;

    radius = 24;

    damage =
      2.5 +
      level * 0.06;

    reward = 4;

    color = [
      105,
      60,
      165
    ];
  }


  enemies.push({

    // ★ 必ず画面のすぐ上から出す
    x: random(
      25,
      width - 25
    ),

    y:
      -radius -
      random(
        20,
        70
      ),

    hp: hp,

    maxHp: hp,

    speed: speed,

    radius: radius,

    damage: damage,

    xp: reward,

    type: type,

    color: color,

    wobble:
      random(TWO_PI),

    slow: 0
  });
}


// =====================================================
// ENEMY UPDATE
// =====================================================

function updateEnemies() {

  for (
    let i =
      enemies.length - 1;
    i >= 0;
    i--
  ) {

    let e =
      enemies[i];


    let angle =
      atan2(
        castle.y - e.y,
        castle.x - e.x
      );


    let speed =
      e.speed;


    if (
      e.slow > 0
    ) {

      speed *= 0.45;

      e.slow -=
        deltaTime;
    }


    e.x +=
      cos(angle) *
      speed;

    e.y +=
      sin(angle) *
      speed;


    // slight wobble

    e.x +=
      sin(
        frameCount *
        0.025 +
        e.wobble
      ) *
      0.08;


    // castle contact

    if (
      dist(
        e.x,
        e.y,
        castle.x,
        castle.y
      ) <
      e.radius + 38
    ) {

      castle.hp -=
        e.damage *
        deltaTime /
        1000;
    }
  }
}

// =====================================================
// AUTO SPELL
// =====================================================

function autoSpell() {

  if (
    millis() -
    spellTimer <
    magic.cooldown
  ) {
    return;
  }


  if (
    enemies.length === 0
  ) {
    return;
  }


  spellTimer =
    millis();


  // 城に近い敵を優先

  let targets =
    [...enemies];


  targets.sort(
    (a, b) => {

      let da =
        dist(
          a.x,
          a.y,
          castle.x,
          castle.y
        );

      let db =
        dist(
          b.x,
          b.y,
          castle.x,
          castle.y
        );

      return da - db;
    }
  );


  for (
    let i = 0;
    i < magic.missileCount;
    i++
  ) {

    let target =
      targets[
        i %
        targets.length
      ];


    missiles.push({

      x:
        castle.x,

      y:
        castle.y - 95,

      target:
        target,

      speed:
        magic.missileSpeed,

      damage:
        magic.damage,

      pierce:
        magic.pierce,

      hit: [],

      life: 5000
    });
  }
}


// =====================================================
// MISSILE UPDATE
// =====================================================

function updateMissiles() {

  for (
    let i =
      missiles.length - 1;
    i >= 0;
    i--
  ) {

    let m =
      missiles[i];


    m.life -=
      deltaTime;


    if (
      m.life <= 0
    ) {

      missiles.splice(
        i,
        1
      );

      continue;
    }


    if (
      !enemies.includes(
        m.target
      )
    ) {

      missiles.splice(
        i,
        1
      );

      continue;
    }


    let angle =
      atan2(
        m.target.y - m.y,
        m.target.x - m.x
      );


    m.x +=
      cos(angle) *
      m.speed;

    m.y +=
      sin(angle) *
      m.speed;


    // trail

    particles.push({

      x: m.x,

      y: m.y,

      vx: 0,

      vy: 0,

      life: 100,

      size:
        random(
          3,
          6
        )
    });


    // collision

    let remove =
      false;


    for (
      let j =
        enemies.length - 1;
      j >= 0;
      j--
    ) {

      let e =
        enemies[j];


      if (
        m.hit.includes(e)
      ) {
        continue;
      }


      let d =
        dist(
          m.x,
          m.y,
          e.x,
          e.y
        );


      if (
        d <
        e.radius + 9
      ) {

        m.hit.push(e);


        e.hp -=
          m.damage;


        // ICE

        if (
          magic.ice > 0
        ) {

          e.slow =
            1000 *
            magic.ice;
        }


        // EXPLOSION

        if (
          magic.explosion > 0
        ) {

          damageArea(
            e,
            magic.explosion
          );
        }


        createExplosion(
          e.x,
          e.y,
          10
        );


        // ★ 即時XP

        if (
          e.hp <= 0
        ) {

          killEnemy(e);
        }


        if (
          m.pierce > 0
        ) {

          m.pierce--;

        } else {

          remove = true;

          break;
        }
      }
    }


    if (
      remove
    ) {

      missiles.splice(
        i,
        1
      );
    }
  }
}

// =====================================================
// SPECIAL MAGIC
// =====================================================

function updateSpecialMagic() {

  // LIGHTNING

  if (
    magic.lightning > 0 &&
    enemies.length > 0 &&
    millis() -
    specialTimer >
    1300
  ) {

    specialTimer =
      millis();


    let count =
      min(
        magic.lightning,
        enemies.length
      );


    for (
      let i = 0;
      i < count;
      i++
    ) {

      let e =
        random(
          enemies
        );


      e.hp -=
        25 +
        level * 5;


      effects.push({

        type:
          "lightning",

        x:
          e.x,

        y:
          e.y,

        life:
          300
      });


      if (
        e.hp <= 0
      ) {

        killEnemy(e);
      }
    }
  }


  // METEOR

  if (
    magic.meteor > 0 &&
    enemies.length > 0 &&
    millis() -
    specialTimer >
    5000
  ) {

    specialTimer =
      millis();


    let e =
      random(
        enemies
      );


    effects.push({

      type:
        "meteor",

      x:
        e.x,

      y:
        e.y,

      life:
        900,

      damage:
        90 *
        magic.meteor,

      hit: false
    });
  }


  // TORNADO

  if (
    magic.tornado > 0
  ) {

    for (
      let e of enemies
    ) {

      if (
        random() <
        0.008 *
        magic.tornado
      ) {

        e.y -=
          15 *
          magic.tornado;
      }
    }
  }


  // HOLY

  if (
    magic.holy > 0
  ) {

    for (
      let e of [
        ...enemies
      ]
    ) {

      let d =
        dist(
          e.x,
          e.y,
          castle.x,
          castle.y
        );


      if (
        d < 170 &&
        random() <
        0.007 *
        magic.holy
      ) {

        e.hp -=
          6 *
          magic.holy;


        effects.push({

          type:
            "holy",

          x:
            e.x,

          y:
            e.y,

          life:
            250
        });


        if (
          e.hp <= 0
        ) {

          killEnemy(e);
        }
      }
    }
  }
}


// =====================================================
// AREA DAMAGE
// =====================================================

function damageArea(
  target,
  amount
) {

  for (
    let e of [
      ...enemies
    ]
  ) {

    if (
      e === target
    ) {
      continue;
    }


    let d =
      dist(
        target.x,
        target.y,
        e.x,
        e.y
      );


    if (
      d < 65
    ) {

      e.hp -=
        amount;


      if (
        e.hp <= 0
      ) {

        killEnemy(e);
      }
    }
  }
}


// =====================================================
// KILL
// =====================================================

function killEnemy(e) {

  let index =
    enemies.indexOf(e);


  if (
    index === -1
  ) {
    return;
  }


  createExplosion(
    e.x,
    e.y,
    18
  );


  // ★ 倒した瞬間にXP

  xp +=
    e.xp;


  enemies.splice(
    index,
    1
  );


  // level up

  if (
    xp >= xpNeed
  ) {

    xp -=
      xpNeed;

    levelUp();
  }
}

// =====================================================
// INPUT
// =====================================================

function mousePressed() {

  if (
    gameState !==
    "levelup"
  ) {

    if (
      gameState ===
      "gameover"
    ) {

      restart();
    }

    return;
  }


  let cardX = 18;

  let cardW =
    width - 36;

  let cardH = 88;


  for (
    let i = 0;
    i < upgrades.length;
    i++
  ) {

    let y =
      height * 0.27 +
      i * 108;


    if (
      mouseX >= cardX &&
      mouseX <=
      cardX + cardW &&
      mouseY >= y &&
      mouseY <=
      y + cardH
    ) {

      applyUpgrade(
        upgrades[i]
      );

      gameState =
        "play";

      spellTimer =
        millis();

      break;
    }
  }
}


// =====================================================
// UPGRADE
// =====================================================

function applyUpgrade(u) {

  if (
    u.type ===
    "damage"
  ) {

    magic.damage +=
      25;
  }


  if (
    u.type ===
    "lightning"
  ) {

    magic.lightning++;
  }


  if (
    u.type ===
    "ice"
  ) {

    magic.ice++;
  }


  if (
    u.type ===
    "explosion"
  ) {

    magic.explosion +=
      20;
  }


  if (
    u.type ===
    "pierce"
  ) {

    magic.pierce++;
  }


  if (
    u.type ===
    "cooldown"
  ) {

    magic.cooldown =
      max(
        650,
        magic.cooldown - 220
      );
  }


  if (
    u.type ===
    "meteor"
  ) {

    magic.meteor++;
  }


  if (
    u.type ===
    "tornado"
  ) {

    magic.tornado++;
  }


  if (
    u.type ===
    "holy"
  ) {

    magic.holy++;
  }


  if (
    u.type ===
    "heal"
  ) {

    magic.heal++;
  }


  if (
    u.type ===
    "count"
  ) {

    magic.missileCount++;
  }
}


// =====================================================
// EFFECTS
// =====================================================

function updateEffects() {

  for (
    let i =
      effects.length - 1;
    i >= 0;
    i--
  ) {

    let e =
      effects[i];


    // meteor impact

    if (
      e.type ===
      "meteor" &&
      e.life < 400 &&
      !e.hit
    ) {

      e.hit = true;


      for (
        let enemy of [
          ...enemies
        ]
      ) {

        if (
          dist(
            e.x,
            e.y,
            enemy.x,
            enemy.y
          ) < 80
        ) {

          enemy.hp -=
            e.damage;


          if (
            enemy.hp <= 0
          ) {

            killEnemy(
              enemy
            );
          }
        }
      }


      createExplosion(
        e.x,
        e.y,
        30
      );
    }


    e.life -=
      deltaTime;


    if (
      e.life <= 0
    ) {

      effects.splice(
        i,
        1
      );
    }
  }
}


// =====================================================
// DRAW EFFECTS
// =====================================================

function drawEffects() {

  for (
    let e of effects
  ) {


    if (
      e.type ===
      "lightning"
    ) {

      stroke(
        100,
        220,
        255,
        230
      );

      strokeWeight(4);

      line(
        e.x,
        e.y - 230,
        e.x,
        e.y
      );

      noStroke();
    }


    if (
      e.type ===
      "meteor"
    ) {

      let alpha =
        map(
          e.life,
          0,
          900,
          0,
          210
        );


      fill(
        255,
        80,
        20,
        alpha
      );

      ellipse(
        e.x,
        e.y,
        90
      );


      fill(
        255,
        220,
        70,
        alpha
      );

      ellipse(
        e.x,
        e.y,
        40
      );
    }


    if (
      e.type ===
      "holy"
    ) {

      noFill();

      stroke(
        255,
        240,
        100,
        210
      );

      strokeWeight(4);

      ellipse(
        e.x,
        e.y,
        45
      );

      noStroke();
    }
  }
}


// =====================================================
// PARTICLES
// =====================================================

function createExplosion(
  x,
  y,
  amount
) {

  for (
    let i = 0;
    i < amount;
    i++
  ) {

    let angle =
      random(TWO_PI);

    let speed =
      random(
        1,
        4
      );


    particles.push({

      x: x,

      y: y,

      vx:
        cos(angle) *
        speed,

      vy:
        sin(angle) *
        speed,

      life:
        random(
          250,
          550
        ),

      size:
        random(
          3,
          7
        )
    });
  }
}


function updateParticles() {

  for (
    let i =
      particles.length - 1;
    i >= 0;
    i--
  ) {

    let p =
      particles[i];


    p.x +=
      p.vx;

    p.y +=
      p.vy;


    p.vx *=
      0.97;

    p.vy *=
      0.97;


    p.life -=
      deltaTime;


    if (
      p.life <= 0
    ) {

      particles.splice(
        i,
        1
      );
    }
  }
}


function drawParticles() {

  noStroke();


  for (
    let p of particles
  ) {

    let alpha =
      map(
        p.life,
        0,
        550,
        0,
        220
      );


    fill(
      80,
      210,
      255,
      alpha
    );


    ellipse(
      p.x,
      p.y,
      p.size
    );
  }
}
