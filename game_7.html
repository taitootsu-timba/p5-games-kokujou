function resetBattle(){
  createUnits();
  battleOutcome = null;
  gameState = 'battle';
  nextTurn();

function fmt(n){ return Math.round(n).toLocaleString('ja-JP'); }

/* ===================== 行動順(AV/ゲージ)ロジック ===================== */
function pickNext(units){
  const alive = units.filter(u=>u.alive);
  return alive.reduce((a,b)=> a.av < b.av ? a : b);
}
function advanceAV(u){ u.av += BASE_AV / u.speed; }

function computeUpcoming(units, currentId, count){
  const clones = units.map(u=>({...u}));
  const cur = clones.find(u=>u.id===currentId);
  if(cur) advanceAV(cur);
  const list = [];
  for(let i=0;i<count;i++){
    const alive = clones.filter(u=>u.alive);
    if(alive.length===0) break;
    const nxt = alive.reduce((a,b)=> a.av < b.av ? a : b);
    list.push({name:nxt.name, isAlly:nxt.isAlly});
    advanceAV(nxt);
  }
  return list;
}

function nextTurn(){
  currentActor = pickNext(allUnits);
  upcomingQueue = computeUpcoming(allUnits, currentActor.id, 5);
  if(currentActor.isAlly){
    phase = 'playerInput';
  } else {
    phase = 'enemyWait';
    phaseTimer = millis() + 900;
  }
}

function addPopup(x,y,text,color,big){
  popups.push({x,y,text,color,alpha:255,big:!!big});
}

function playerAttack(type){
  if(phase !== 'playerInput' || gameState !== 'battle') return;
  if(type==='skill' && sp<=0) return; // SPが無ければスキル不可

  const isHealer = (currentActor.id === 'p3'); // 一番右の味方はスキルで回復役

  if(type==='skill' && isHealer){
    // --- 回復スキル：味方全体のHPを500〜750回復 ---
    sp = max(0, sp - 1);
    currentActor.ep = min(100, currentActor.ep + 28);
    const healAmt = floor(random(500, 751));
    for(const p of party){
      if(!p.alive) continue;
      p.hp = min(p.maxHp, p.hp + healAmt);
      addPopup(p.x, p.y-80, '+'+fmt(healAmt), [140,255,150], false);
    }
  } else {
    let dmg;
    if(type==='normal'){
      dmg = 1000000 + floor(random(-250000, 250001));
      if(sp < MAX_SP) sp += 1; // SPが5未満のときのみ回復
      currentActor.ep = min(100, currentActor.ep + 18);
    } else {
      dmg = 1000000 + floor(random(-250000, 250001));
      sp = max(0, sp - 1); // スキル使用でSPを1消費
      currentActor.ep = min(100, currentActor.ep + 28);
    }
    dmg = max(0, dmg);
    damageBoss(dmg);
    hitFlashBoss = 255;
    addPopup(boss.x, boss.y-70, '-'+fmt(dmg), type==='skill'?[255,221,120]:[255,255,255], type==='skill');
    lastBossDamage = { text:'-'+fmt(dmg), skill:(type==='skill') };
  }

  advanceAV(currentActor);
  phase = 'resolve';
  phaseTimer = millis() + 650;
}

function damageBoss(dmg){
  boss.hp = boss.hp - dmg;
  if(boss.hp <= 0){
    if(boss.lives > 1){
      boss.lives -= 1;
      boss.hp = boss.maxHp; // 残機が残っていればHP全回復して続行
    } else {
      boss.lives = 0;
      boss.hp = 0;
      boss.alive = false;
    }
  }
}

function enemyAttack(){
  const aliveAllies = party.filter(p=>p.alive);
  if(aliveAllies.length===0){ phase='resolve'; phaseTimer=millis()+200; return; }
  const target = random(aliveAllies);
  const dmg = max(0, 1000 + floor(random(-500, 501)));
  target.hp = max(0, target.hp - dmg);
  if(target.hp<=0) target.alive = false;
  target.ep = min(100, target.ep + 10);
  hitFlashAlly[target.id] = 255;
  addPopup(target.x, target.y-80, '-'+fmt(dmg), [255,90,90], false);
  advanceAV(currentActor);
  phase = 'resolve';
  phaseTimer = millis() + 650;
}

function checkBattleEnd(){
  if(!boss.alive || boss.hp<=0){
    battleOutcome = 'win';
    gameState = 'end';
    return true;
  }
  if(party.every(p=>!p.alive)){
    battleOutcome = 'lose';
    gameState = 'end';
    return true;
  }
  return false;
}

/* ===================== メインループ ===================== */
function draw(){
  if(gameState === 'title'){
    drawTitle();
  } else if(gameState === 'battle'){
    drawBattle();
    if(phase==='enemyWait' && millis()>=phaseTimer){
      enemyAttack();
    } else if(phase==='resolve' && millis()>=phaseTimer){
      const ended = checkBattleEnd();
      if(!ended) nextTurn();
    }
  } else if(gameState === 'end'){
    drawBattle(true);
    drawEndOverlay();
  }
}

function updateSmoke(){
  if(frameCount % 8 === 0){
    smokeParticles.push({x: 650+65*cos(radians(-7))-40*sin(radians(-7)), y: 355-70+30, r: random(8,14), alpha:160, vy:random(0.6,1.1), vx:random(-0.3,0.3)});
  }
  for(const s of smokeParticles){
    s.y -= s.vy;
    s.x += s.vx + 0.15;
    s.r += 0.25;
    s.alpha -= 1.6;
  }
  smokeParticles = smokeParticles.filter(s=>s.alpha>0);
}
function drawSmoke(){
  noStroke();
  for(const s of smokeParticles){
    fill(210,210,220,s.alpha*0.5);
    circle(s.x, s.y, s.r);
  }
}

function drawHPBar(x,y,w,h,ratio,frontColor,showPercent){
  ratio = constrain(ratio,0,1);
  push();
  rectMode(CORNER);
  noStroke();
  fill(20,15,25,220);
  rect(x,y,w,h,h/2);

  let col;
  if(ratio > 0.5) col = lerpColor(color(255,205,90), color(frontColor[0],frontColor[1],frontColor[2]), (ratio-0.5)*2);
  else col = lerpColor(color(220,60,60), color(255,205,90), ratio*2);

  fill(col);
  if(ratio>0.02) rect(x,y,w*ratio,h,h/2);

  noFill();
  stroke(232,200,116,200);
  strokeWeight(1.5);
  rect(x,y,w,h,h/2);

  if(showPercent){
    noStroke();
    fill(255);
    textAlign(CENTER,CENTER);
    textSize(12);
    text(floor(ratio*100)+' %', x+w/2, y+h/2+1);
  }
  pop();
}

function drawEPGauge(x,y,r,ratio){
  ratio = constrain(ratio,0,1);
  push();
  noStroke();
  fill(15,12,22,230);
  circle(x,y,r*2);

  drawingContext.save();
  drawingContext.beginPath();
  drawingContext.arc(x,y,r-1.5,0,Math.PI*2);
  drawingContext.clip();
  const fillH = (r*2) * ratio;
  noStroke();
  fill(120,205,255);
  rectMode(CORNER);
  rect(x-r, y+r-fillH, r*2, fillH+2);
  drawingContext.restore();

  noFill();
  stroke(180,220,255,220);
  strokeWeight(2);
  circle(x,y,r*2);

  noStroke();
  fill(255);
  textAlign(CENTER,CENTER);
  textSize(9);
  text(floor(ratio*100), x, y+1);

  fill(180,220,255,200);
  textSize(8);
  text('EP', x, y-r-7);
  pop();
}

    fill(255,140,140,220);
    textAlign(CENTER,CENTER);
    textSize(10);
    text('SP不足', btnSkill.x+btnSkill.w/2, btnSkill.y+btnSkill.h+12);
  }

  if(!active){
    noStroke();
    fill(200,200,210,180);
    textAlign(CENTER,CENTER);
    textSize(11);
    text(currentActor ? (currentActor.name+' の行動中…') : '', cmdPanel.x+cmdPanel.w/2+30, cmdPanel.y+cmdPanel.h-12);
  }
  pop();
}

function drawButton(b, label, active, accent){
  push();
  rectMode(CORNER);
  if(active){
    fill(accent[0],accent[1],accent[2],60+30*sin(frameCount*0.1));
    stroke(accent[0],accent[1],accent[2]);
  } else {
    fill(40,38,50,150);
    stroke(90,90,100);
  }
  strokeWeight(2);
  rect(b.x,b.y,b.w,b.h,8);
  noStroke();
  fill(active?255:150);
  textAlign(CENTER,CENTER);
  textSize(17);
  text(label, b.x+b.w/2, b.y+b.h/2+1);
  pop();
}

function updatePopups(){
  for(const p of popups){
    p.y -= 0.7;
    p.alpha -= 4.5;
  }
  popups = popups.filter(p=>p.alpha>0);
}
function drawPopups(){
  push();
  textAlign(CENTER,CENTER);
  for(const p of popups){
    textSize(p.big?28:20);
    stroke(0,0,0,p.alpha);
    strokeWeight(3);
    fill(p.color[0],p.color[1],p.color[2],p.alpha);
    text(p.text, p.x, p.y);
  }
  pop();
}

/* ===================== 戦闘終了画面 ===================== */
function drawSunMandala(x,y,size){
  push();
  translate(x,y);
  rotate(frameCount*0.003);
  noFill();
  for(let r= size*0.4; r<size; r+=size*0.15){
    stroke(232,200,116, 40);
    strokeWeight(1.5);
    circle(0,0,r*2);
  }
  const rays = 28;
  for(let i=0;i<rays;i++){
    const ang = TWO_PI*i/rays;
    push();
    rotate(ang);
    const len = size*(0.55 + 0.15*sin(frameCount*0.02+i));
    stroke(232,200,116, 90);
    strokeWeight(3);
    line(size*0.35,0,len,0);
    pop();
  }
  noStroke();
  fill(232,200,116,25);
  circle(0,0,size*0.9);
  pop();
}

/* ===================== 入力処理 ===================== */
function inRect(mx,my,r){
  return mx>=r.x && mx<=r.x+r.w && my>=r.y && my<=r.y+r.h;
}

function mousePressed(){
  if(gameState==='title'){
    resetBattle();
    return;
  }
  if(gameState==='battle' && phase==='playerInput'){
    if(inRect(mouseX,mouseY,btnNormal)) playerAttack('normal');
    else if(inRect(mouseX,mouseY,btnSkill)) playerAttack('skill');
    return;
  }
  if(gameState==='end'){
    if(inRect(mouseX,mouseY,btnRetry)) resetBattle();
    else if(inRect(mouseX,mouseY,btnExit)) gameState='title';
    return;
  }
}
