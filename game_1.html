const WORLD_SIZE = 5000;
const FOOD_COUNT = 650;
const ENEMY_COUNT = 12;

let player;
let foods = [];
let enemies = [];

function setup() {
  createCanvas(windowWidth, windowHeight);

  player = new Snake(
    0,
    0,
    color(50, 230, 100),
    true
  );

  for (let i = 0; i < FOOD_COUNT; i++) {
    spawnFood();
  }

  for (let i = 0; i < ENEMY_COUNT; i++) {
    spawnEnemy();
  }
}

function draw() {
  background(13, 15, 21);

  if (!player.dead) {
    player.update();
  }

  for (let enemy of enemies) {
    if (!enemy.dead) {
      enemy.update();
    }
  }

  checkCollisions();

  push();

  translate(
    width / 2 - player.x,
    height / 2 - player.y
  );

  drawWorld();

  for (let food of foods) {
    food.show();
  }

  for (let enemy of enemies) {
    if (!enemy.dead) {
      enemy.show();
    }
  }

  player.show();

  pop();

  drawUI();

  // 死んだ敵を削除
  for (let i = enemies.length - 1; i >= 0; i--) {
    if (enemies[i].dead) {
      enemies.splice(i, 1);
    }
  }

  // 敵を補充
  while (enemies.length < ENEMY_COUNT) {
    spawnEnemy();
  }
}

function drawWorld() {
  const half = WORLD_SIZE / 2;

  stroke(27, 30, 38);
  strokeWeight(1);

  for (let x = -half; x <= half; x += 100) {
    line(x, -half, x, half);
  }

  for (let y = -half; y <= half; y += 100) {
    line(-half, y, half, y);
  }

  noFill();
  stroke(90, 100, 120);
  strokeWeight(8);

  rect(
    -half,
    -half,
    WORLD_SIZE,
    WORLD_SIZE
  );
}

// =================================================
// COLLISION
// =================================================

function checkCollisions() {
  if (player.dead) return;

  // -----------------------------------------------
  // プレイヤーの頭 VS 敵の体
  // -----------------------------------------------

  for (let enemy of enemies) {
    if (enemy.dead) continue;

    for (let i = 7; i < enemy.body.length; i++) {
      const p = enemy.body[i];

      const d = dist(
        player.x,
        player.y,
        p.x,
        p.y
      );

      if (
        d <
        player.radius +
        enemy.radius * 0.75
      ) {
        player.die();
        return;
      }
    }
  }

  // -----------------------------------------------
  // 敵の頭 VS プレイヤーの体
  // -----------------------------------------------

  for (let enemy of enemies) {
    if (enemy.dead) continue;

    for (let i = 8; i < player.body.length; i++) {
      const p = player.body[i];

      const d = dist(
        enemy.x,
        enemy.y,
        p.x,
        p.y
      );

      if (
        d <
        enemy.radius +
        player.radius * 0.75
      ) {
        enemy.die();
        break;
      }
    }
  }

  // -----------------------------------------------
  // 頭 VS 頭
  // -----------------------------------------------

  for (let enemy of enemies) {
    if (enemy.dead) continue;

    const d = dist(
      player.x,
      player.y,
      enemy.x,
      enemy.y
    );

    if (
      d <
      player.radius +
      enemy.radius
    ) {
      if (
        player.body.length >=
        enemy.body.length
      ) {
        enemy.die();
      } else {
        player.die();
        return;
      }
    }
  }

  // -----------------------------------------------
  // 敵同士の衝突
  // -----------------------------------------------

  for (let i = 0; i < enemies.length; i++) {
    const a = enemies[i];

    if (a.dead) continue;

    for (let j = i + 1; j < enemies.length; j++) {
      const b = enemies[j];

      if (b.dead) continue;

      // Aの頭 VS Bの体
      if (
        headHitsBody(a, b)
      ) {
        a.die();
      }

      // Bの頭 VS Aの体
      if (
        headHitsBody(b, a)
      ) {
        b.die();
      }
    }
  }
}

function headHitsBody(head, bodySnake) {
  for (
    let i = 7;
    i < bodySnake.body.length;
    i++
  ) {
    const p = bodySnake.body[i];

    const d = dist(
      head.x,
      head.y,
      p.x,
      p.y
    );

    if (
      d <
      head.radius +
      bodySnake.radius * 0.75
    ) {
      return true;
    }
  }

  return false;
}

class Food {
  constructor(x, y) {
    this.x = x;
    this.y = y;
    this.r = random(4, 9);

    this.col = color(
      random(80, 255),
      random(80, 255),
      random(80, 255)
    );
  }

  show() {
    noStroke();

    fill(
      red(this.col),
      green(this.col),
      blue(this.col),
      40
    );

    circle(
      this.x,
      this.y,
      this.r * 4
    );

    fill(this.col);

    circle(
      this.x,
      this.y,
      this.r * 2
    );
  }
}

// =================================================
// SNAKE
// =================================================

class Snake {
  constructor(x, y, col, isPlayer) {
    this.x = x;
    this.y = y;

    this.col = col;
    this.isPlayer = isPlayer;

    this.angle = random(TWO_PI);

    this.radius = 13;

    this.normalSpeed =
      isPlayer ? 3.0 : random(2.6, 3.2);

    this.boostSpeed =
      isPlayer ? 6.2 : 4.2;

    this.speed = this.normalSpeed;

    this.targetLength =
      isPlayer
        ? 40
        : floor(random(35, 100));

    this.boostEnergy = 1;

    this.dead = false;

    this.aiTimer = random(30, 100);

    this.body = [];

    for (
      let i = 0;
      i < this.targetLength;
      i++
    ) {
      this.body.push({
        x:
          this.x -
          cos(this.angle) * i * 7,

        y:
          this.y -
          sin(this.angle) * i * 7
      });
    }
  }

  // =================================================
  // UPDATE
  // =================================================

  update() {
    if (this.dead) return;

    if (this.isPlayer) {
      this.updatePlayer();
    } else {
      this.updateAI();
    }

    // 移動
    this.x +=
      cos(this.angle) *
      this.speed;

    this.y +=
      sin(this.angle) *
      this.speed;

    // 頭を追加
    this.body.unshift({
      x: this.x,
      y: this.y
    });

    // 体を切る
    while (
      this.body.length >
      this.targetLength
    ) {
      this.body.pop();
    }

    // 壁
    const half =
      WORLD_SIZE / 2 - 30;

    if (
      this.x < -half ||
      this.x > half ||
      this.y < -half ||
      this.y > half
    ) {
      this.die();
      return;
    }

    this.eatFood();
  }

  // =================================================
  // PLAYER
  // =================================================

  updatePlayer() {
    const targetAngle = atan2(
      mouseY - height / 2,
      mouseX - width / 2
    );

    const diff =
      angleDifference(
        targetAngle,
        this.angle
      );

    this.angle +=
      diff * 0.13;

    const boosting =
      mouseIsPressed ||
      keyIsDown(32);

    if (
      boosting &&
      this.boostEnergy > 0.01 &&
      this.targetLength > 20
    ) {
      this.speed =
        this.boostSpeed;

      this.boostEnergy -= 0.004;

      // ブースト中は体を消費
      if (frameCount % 5 === 0) {
        this.targetLength--;

        if (this.body.length > 5) {
          const tail =
            this.body[
              this.body.length - 1
            ];

          foods.push(
            new Food(
              tail.x + random(-6, 6),
              tail.y + random(-6, 6)
            )
          );
        }
      }
    } else {
      this.speed =
        this.normalSpeed;

      this.boostEnergy +=
        0.0025;
    }

    this.boostEnergy =
      constrain(
        this.boostEnergy,
        0,
        1
      );
  }

  // =================================================
  // 強化AI
  // =================================================

  updateAI() {
    let closestDistance =
      Infinity;

    let closestX =
      this.x;

    let closestY =
      this.y;

    // -----------------------------------------------
    // プレイヤーとの距離
    // -----------------------------------------------

    let playerDistance =
      dist(
        this.x,
        this.y,
        player.x,
        player.y
      );

    // プレイヤーの未来位置を予測
    let prediction =
      constrain(
        playerDistance / 50,
        4,
        16
      );

    let predictedX =
      player.x +
      cos(player.angle) *
      player.speed *
      prediction;

    let predictedY =
      player.y +
      sin(player.angle) *
      player.speed *
      prediction;

    // -----------------------------------------------
    // 基本AI
    // -----------------------------------------------

    if (
      playerDistance < 1400 &&
      !player.dead
    ) {
      closestX =
        predictedX;

      closestY =
        predictedY;

      closestDistance =
        playerDistance;
    }

    // -----------------------------------------------
    // プレイヤーを横から挟む
    // -----------------------------------------------

    if (
      playerDistance < 800 &&
      playerDistance > 200 &&
      !player.dead
    ) {
      const side =
        sin(
          player.angle -
          this.angle
        );

      const sideDirection =
        side >= 0
          ? 1
          : -1;

      const surroundAngle =
        player.angle +
        HALF_PI *
        sideDirection;

      closestX =
        player.x +
        cos(surroundAngle) *
        220;

      closestY =
        player.y +
        sin(surroundAngle) *
        220;
    }

    // -----------------------------------------------
    // プレイヤーの体を狙う
    // -----------------------------------------------

    if (
      playerDistance < 650 &&
      !player.dead
    ) {
      let index =
        floor(
          map(
            playerDistance,
            0,
            650,
            12,
            min(
              player.body.length - 1,
              60
            )
          )
        );

      index =
        constrain(
          index,
          8,
          player.body.length - 1
        );

      const p =
        player.body[index];

      if (p) {
        closestX =
          p.x;

        closestY =
          p.y;
      }
    }

    // -----------------------------------------------
    // 他の敵を避ける
    // -----------------------------------------------

    for (let other of enemies) {
      if (
        other === this ||
        other.dead
      ) {
        continue;
      }

      const d =
        dist(
          this.x,
          this.y,
          other.x,
          other.y
        );

      if (d < 260) {
        // 相手が大きいなら逃げる
        if (
          other.body.length >
          this.body.length * 1.15
        ) {
          const escape =
            atan2(
              this.y - other.y,
              this.x - other.x
            );

          const diff =
            angleDifference(
              escape,
              this.angle
            );

          this.angle +=
            diff * 0.1;
        }
      }
    }

    // -----------------------------------------------
    // ターゲットへ曲がる
    // -----------------------------------------------

    if (
      closestDistance <
      Infinity
    ) {
      const target =
        atan2(
          closestY - this.y,
          closestX - this.x
        );

      const diff =
        angleDifference(
          target,
          this.angle
        );

      let turnPower =
        0.035;

      if (
        playerDistance < 700
      ) {
        turnPower =
          0.06;
      }

      if (
        playerDistance < 350
      ) {
        turnPower =
          0.10;
      }

      this.angle +=
        diff * turnPower;
    } else {
      // プレイヤーが遠いときも
      // ランダムに動く
      this.aiTimer--;

      if (this.aiTimer <= 0) {
        this.angle +=
          random(-0.8, 0.8);

        this.aiTimer =
          random(40, 140);
      }
    }

    // -----------------------------------------------
    // 壁回避
    // -----------------------------------------------

    const margin = 400;

    if (
      this.x <
      -WORLD_SIZE / 2 +
      margin
    ) {
      this.angle =
        lerpAngle(
          this.angle,
          0,
          0.10
        );
    }

    if (
      this.x >
      WORLD_SIZE / 2 -
      margin
    ) {
      this.angle =
        lerpAngle(
          this.angle,
          PI,
          0.10
        );
    }

    if (
      this.y <
      -WORLD_SIZE / 2 +
      margin
    ) {
      this.angle =
        lerpAngle(
          this.angle,
          HALF_PI,
          0.10
        );
    }

    if (
      this.y >
      WORLD_SIZE / 2 -
      margin
    ) {
      this.angle =
        lerpAngle(
          this.angle,
          -HALF_PI,
          0.10
        );
    }

    // -----------------------------------------------
    // 接近時は加速
    // -----------------------------------------------

    if (
      playerDistance < 550 &&
      !player.dead &&
      this.targetLength > 25
    ) {
      this.speed =
        this.boostSpeed;

      // 敵も少し体を消費
      if (
        frameCount % 7 === 0
      ) {
        this.targetLength--;

        if (
          this.body.length > 10
        ) {
          const tail =
            this.body[
              this.body.length - 1
            ];

          foods.push(
            new Food(
              tail.x,
              tail.y
            )
          );
        }
      }
    } else {
      this.speed =
        this.normalSpeed;
    }
  }

  // =================================================
  // FOOD
  // =================================================

  eatFood() {
    for (
      let i = foods.length - 1;
      i >= 0;
      i--
    ) {
      const food =
        foods[i];

      const d =
        dist(
          this.x,
          this.y,
          food.x,
          food.y
        );

      if (
        d <
        this.radius +
        food.r
      ) {
        this.targetLength += 2;

        foods.splice(i, 1);

        spawnFood();
      }
    }
  }

  // =================================================
  // DRAW
  // =================================================

  show() {
    noStroke();

    // 体
    for (
      let i =
        this.body.length - 1;
      i >= 0;
      i--
    ) {
      const p =
        this.body[i];

      let size =
        this.radius * 2;

      const ratio =
        i / this.body.length;

      size *=
        1 -
        ratio * 0.18;

      fill(this.col);

      circle(
        p.x,
        p.y,
        size
      );
    }

    // 頭
    fill(this.col);

    circle(
      this.x,
      this.y,
      this.radius * 2.2
    );

    this.drawEyes();
  }

  // =================================================
  // EYES
  // =================================================

  drawEyes() {
    const fx =
      cos(this.angle);

    const fy =
      sin(this.angle);

    const sx =
      -fy;

    const sy =
      fx;

    const eyeDistance = 6;

    const x1 =
      this.x +
      fx * 5 +
      sx * eyeDistance;

    const y1 =
      this.y +
      fy * 5 +
      sy * eyeDistance;

    const x2 =
      this.x +
      fx * 5 -
      sx * eyeDistance;

    const y2 =
      this.y +
      fy * 5 -
      sy * eyeDistance;

    fill(255);

    circle(
      x1,
      y1,
      9
    );

    circle(
      x2,
      y2,
      9
    );

    fill(20);

    circle(
      x1 + fx * 2,
      y1 + fy * 2,
      4
    );

    circle(
      x2 + fx * 2,
      y2 + fy * 2,
      4
    );
  }

  // =================================================
  // DIE
  // =================================================

  die() {
    if (this.dead) return;

    this.dead = true;

    // プレイヤー
    if (this.isPlayer) {
      return;
    }

    // 敵の体をエサにする
    for (
      let i = 0;
      i < this.body.length;
      i += 3
    ) {
      const p =
        this.body[i];

      foods.push(
        new Food(
          p.x + random(-8, 8),
          p.y + random(-8, 8)
        )
      );
    }
  }
}

// =================================================
// SPAWN FOOD
// =================================================

function spawnFood() {
  foods.push(
    new Food(
      random(
        -WORLD_SIZE / 2 + 50,
        WORLD_SIZE / 2 - 50
      ),
      random(
        -WORLD_SIZE / 2 + 50,
        WORLD_SIZE / 2 - 50
      )
    )
  );
}

// =================================================
// SPAWN ENEMY
// =================================================

function spawnEnemy() {
  let x;
  let y;

  do {
    x =
      random(
        -WORLD_SIZE / 2 + 300,
        WORLD_SIZE / 2 - 300
      );

    y =
      random(
        -WORLD_SIZE / 2 + 300,
        WORLD_SIZE / 2 - 300
      );
  } while (
    dist(
      x,
      y,
      player.x,
      player.y
    ) < 700
  );

  const col =
    color(
      random(60, 255),
      random(60, 255),
      random(60, 255)
    );

  enemies.push(
    new Snake(
      x,
      y,
      col,
      false
    )
  );
}

// =================================================
// ANGLE
// =================================================

function angleDifference(a, b) {
  let d = a - b;

  while (d > PI) {
    d -= TWO_PI;
  }

  while (d < -PI) {
    d += TWO_PI;
  }

  return d;
}

function lerpAngle(a, b, amount) {
  return (
    a +
    angleDifference(b, a) *
    amount
  );
}

// =================================================
// RESTART
// =================================================

function mousePressed() {
  if (player.dead) {
    restartGame();
  }
}

function keyPressed() {
  if (
    player.dead &&
    keyCode === ENTER
  ) {
    restartGame();
  }
}

function restartGame() {
  player =
    new Snake(
      0,
      0,
      color(50, 230, 100),
      true
    );

  foods = [];

  for (
    let i = 0;
    i < FOOD_COUNT;
    i++
  ) {
    spawnFood();
  }

  enemies = [];

  for (
    let i = 0;
    i < ENEMY_COUNT;
    i++
  ) {
    spawnEnemy();
  }
}

function drawUI() {
  noStroke();

  textAlign(
    LEFT,
    TOP
  );

  fill(255);

  textSize(20);

  text(
    "長さ: " +
    player.body.length,
    20,
    20
  );

  textSize(14);

  fill(190);

  text(
    "マウス：移動　｜　クリック / SPACE：加速",
    20,
    50
  );

  // ブーストゲージ
  const gaugeW = 220;
  const gaugeH = 15;

  fill(40);

  rect(
    20,
    80,
    gaugeW,
    gaugeH,
    8
  );

  fill(
    255,
    210,
    40
  );

  rect(
    20,
    80,
    gaugeW *
      player.boostEnergy,
    gaugeH,
    8
  );

  // ランキング
  let ranking =
    enemies
      .filter(e => !e.dead)
      .concat([player])
      .sort(
        (a, b) =>
          b.body.length -
          a.body.length
      );

  fill(255);
  textSize(16);

  text(
    "RANKING",
    width - 180,
    20
  );

  for (
    let i = 0;
    i < min(5, ranking.length);
    i++
  ) {
    const snake =
      ranking[i];

    fill(
      snake.isPlayer
        ? color(80, 255, 100)
        : 255
    );

    text(
      (i + 1) +
      ". " +
      snake.body.length,
      width - 180,
      45 + i * 22
    );
  }

  // GAME OVER
  if (player.dead) {
    fill(
      255,
      60,
      70
    );

    textAlign(
      CENTER,
      CENTER
    );

    textSize(60);

    text(
      "GAME OVER",
      width / 2,
      height / 2 - 50
    );

    fill(255);

    textSize(20);

    text(
      "クリックまたは ENTER でリスタート",
      width / 2,
      height / 2 + 20
    );
  }
}
