// ==========================================
// ZOMBIE SURVIVAL - p5.js
// ==========================================

let zombies = [];
let bullets = [];
let particles = [];
let casings = [];
let bloodSplats = [];

let score = 0;
let hp = 100;

let ammo = 30;
const maxAmmo = 30;

let reloading = false;
let reloadTimer = 0;

let shake = 0;
let muzzleFlash = 0;

let gameStarted = false;
let gameOver = false;

let spawnTimer = 0;
let difficulty = 1;

function setup() {
  createCanvas(900, 650);
  angleMode(RADIANS);
  textFont("Arial");
}

function draw() {
  background(12, 15, 14);

  if (!gameStarted) {
    drawTitle();
    return;
  }

  if (gameOver) {
    drawGameOver();
    return;
  }

  push();

  // 画面揺れ
  if (shake > 0) {
    translate(random(-shake, shake), random(-shake, shake));
    shake *= 0.88;
  }

  drawWorld();

  updateZombies();
  updateBullets();
  updateParticles();
  updateCasings();

  drawPlayer();
  drawCrosshair();

  pop();

  drawHUD();

  // ゾンビ出現
  spawnTimer--;

  if (spawnTimer <= 0) {
    spawnZombie();
    spawnTimer = max(18, 65 - difficulty * 4);
  }

  // 時間経過で少しずつ難しくなる
  difficulty += 0.002;

  if (muzzleFlash > 0) {
    muzzleFlash--;
  }

  // リロード
  if (reloading) {
    reloadTimer--;

    if (reloadTimer <= 0) {
      ammo = maxAmmo;
      reloading = false;
    }
  }
}

function drawTitle() {
  background(8, 10, 9);

  for (let i = 0; i < 20; i++) {
    noStroke();
    fill(80, 0, 0, 10);

    ellipse(
      random(width),
      random(height),
      random(100, 400)
    );
  }

  textAlign(CENTER, CENTER);

  fill(180, 20, 20);
  textSize(72);
  text("ZOMBIE", width / 2, height / 2 - 80);

  fill(230);
  textSize(55);
  text("SURVIVAL", width / 2, height / 2 - 15);

  fill(170);
  textSize(20);
  text("CLICK TO START", width / 2, height / 2 + 70);

  fill(100);
  textSize(14);
  text(
    "CLICK : SHOOT    R : RELOAD",
    width / 2,
    height / 2 + 110
  );

  textAlign(LEFT, BASELINE);
}

// 背景

function drawWorld() {
  noStroke();

  fill(27, 30, 27);
  rect(0, 0, width, height);

  fill(34, 35, 31);
  rect(80, 80, width - 160, height - 160);

  // 床のタイル
  stroke(45, 46, 41);
  strokeWeight(1);

  for (let x = 80; x < width - 80; x += 45) {
    line(x, 80, x, height - 80);
  }

  for (let y = 80; y < height - 80; y += 45) {
    line(80, y, width - 80, y);
  }

  // 壁
  noStroke();
  fill(18, 20, 19);

  rect(0, 0, width, 80);
  rect(0, height - 80, width, 80);
  rect(0, 0, 80, height);
  rect(width - 80, 0, 80, height);

  // 霧
  for (let i = 0; i < 10; i++) {
    fill(150, 160, 150, 4);

    ellipse(
      random(width),
      random(height),
      random(100, 350),
      random(50, 180)
    );
  }

  // 血痕
  for (let s of bloodSplats) {
    noStroke();
    fill(100, 5, 5, 150);

    ellipse(
      s.x,
      s.y,
      s.size,
      s.size * 0.5
    );
  }
}

function drawPlayer() {
  let angle = atan2(
    mouseY - height / 2,
    mouseX - width / 2
  );

  push();

  translate(width / 2, height / 2);
  rotate(angle);

  // 腕
  fill(150, 105, 75);
  noStroke();

  ellipse(10, -12, 70, 18);
  ellipse(10, 12, 70, 18);

  // 銃
  fill(35);
  rect(10, -8, 70, 16, 3);

  // 銃身
  fill(15);
  rect(65, -4, 35, 8);

  // グリップ
  fill(45);
  rect(25, 7, 18, 35, 4);

  pop();
}

// ゾンビ
// ==========================================

function spawnZombie() {
  let side = floor(random(4));
  let x;
  let y;

  if (side === 0) {
    x = random(width);
    y = -50;
  } else if (side === 1) {
    x = width + 50;
    y = random(height);
  } else if (side === 2) {
    x = random(width);
    y = height + 50;
  } else {
    x = -50;
    y = random(height);
  }

  let type = random();

  zombies.push({
    x: x,
    y: y,

    // ★ 少し小さくした
    size: type < 0.2 ? 55 : 40,

    // ★ 少し速くした
    speed:
      type < 0.2
        ? random(1.0, 1.4) * difficulty
        : random(0.7, 1.0) * difficulty,

    // 大型ゾンビは3発
    hp:
      type < 0.2
        ? 3
        : 1,

    maxHP:
      type < 0.2
        ? 3
        : 1,

    wobble: random(TWO_PI),
    hitFlash: 0
  });
}

function updateZombies() {
  for (let i = zombies.length - 1; i >= 0; i--) {
    let z = zombies[i];

    let dx = width / 2 - z.x;
    let dy = height / 2 - z.y;

    let angle = atan2(dy, dx);

    // 少しふらつきながら進む
    z.wobble += 0.04;

    let wobbleX = sin(z.wobble) * 0.4;
    let wobbleY = cos(z.wobble * 1.3) * 0.4;

    z.x += cos(angle) * z.speed + wobbleX;
    z.y += sin(angle) * z.speed + wobbleY;

    drawZombie(z);

    // プレイヤーに接触
    if (
      dist(
        z.x,
        z.y,
        width / 2,
        height / 2
      ) < z.size * 0.45 + 25
    ) {
      hp -= z.size > 50 ? 0.7 : 0.4;

      shake = 3;

      if (hp <= 0) {
        hp = 0;
        gameOver = true;
      }
    }

    if (z.hitFlash > 0) {
      z.hitFlash--;
    }
  }
}

// ==========================================
// ゾンビ描画
// ==========================================

function drawZombie(z) {
  push();

  translate(z.x, z.y);

  let angle = atan2(
    height / 2 - z.y,
    width / 2 - z.x
  );

  rotate(angle + HALF_PI);

  // 影
  noStroke();

  fill(0, 0, 0, 90);

  ellipse(
    0,
    z.size * 0.42,
    z.size * 0.9,
    z.size * 0.25
  );

  // 足
  stroke(35, 40, 35);
  strokeWeight(z.size * 0.16);

  line(
    -z.size * 0.18,
    z.size * 0.2,
    -z.size * 0.25,
    z.size * 0.55
  );

  line(
    z.size * 0.18,
    z.size * 0.2,
    z.size * 0.25,
    z.size * 0.55
  );

  // 腕
  stroke(75, 85, 65);
  strokeWeight(z.size * 0.13);

  line(
    -z.size * 0.25,
    -z.size * 0.05,
    -z.size * 0.55,
    z.size * 0.12
  );

  line(
    z.size * 0.25,
    -z.size * 0.05,
    z.size * 0.55,
    z.size * 0.12
  );

  // 体
  noStroke();

  fill(
    z.hitFlash > 0 ? 230 : 70,
    z.hitFlash > 0 ? 40 : 80,
    z.hitFlash > 0 ? 40 : 65
  );

  ellipse(
    0,
    z.size * 0.05,
    z.size * 0.58,
    z.size * 0.7
  );

  // 首
  fill(80, 90, 70);

  rect(
    -z.size * 0.1,
    -z.size * 0.35,
    z.size * 0.2,
    z.size * 0.18
  );

  // 頭
  fill(
    z.hitFlash > 0 ? 255 : 100,
    z.hitFlash > 0 ? 50 : 110,
    z.hitFlash > 0 ? 50 : 80
  );

  ellipse(
    0,
    -z.size * 0.5,
    z.size * 0.55,
    z.size * 0.55
  );

  // 髪
  fill(25);

  arc(
    0,
    -z.size * 0.58,
    z.size * 0.5,
    z.size * 0.35,
    PI,
    TWO_PI
  );

  // 目
  fill(220, 20, 20);

  ellipse(
    -z.size * 0.12,
    -z.size * 0.52,
    z.size * 0.07,
    z.size * 0.06
  );

  ellipse(
    z.size * 0.12,
    -z.size * 0.52,
    z.size * 0.07,
    z.size * 0.06
  );

  // 口
  stroke(35);
  strokeWeight(2);

  line(
    -z.size * 0.12,
    -z.size * 0.39,
    z.size * 0.12,
    -z.size * 0.39
  );

  pop();

  // HPバー
  if (z.hp < z.maxHP) {
    noStroke();

    fill(30);

    rect(
      z.x - 25,
      z.y - z.size * 0.9,
      50,
      5
    );

    fill(210, 30, 30);

    rect(
      z.x - 25,
      z.y - z.size * 0.9,
      50 * (z.hp / z.maxHP),
      5
    );
  }
}

// ==========================================
// 射撃
// ==========================================

function shoot() {
  if (
    reloading ||
    ammo <= 0 ||
    gameOver
  ) {
    return;
  }

  ammo--;

  let angle = atan2(
    mouseY - height / 2,
    mouseX - width / 2
  );

  // 少しだけ弾が散る
  angle += random(-0.025, 0.025);

  bullets.push({
    x: width / 2 + cos(angle) * 90,
    y: height / 2 + sin(angle) * 90,

    vx: cos(angle) * 18,
    vy: sin(angle) * 18,

    life: 70
  });

  // 発射演出
  muzzleFlash = 4;

  // 画面揺れ
  shake = 5;

  // 薬莢
  casings.push({
    x: width / 2,
    y: height / 2,
    vx: random(-3, 3),
    vy: random(-3, 3),
    life: 60
  });

  // 火花
  for (let i = 0; i < 8; i++) {
    particles.push({
      x: width / 2 + cos(angle) * 80,
      y: height / 2 + sin(angle) * 80,

      vx: cos(angle) * random(2, 6),
      vy: sin(angle) * random(2, 6),

      life: 30,

      color: [255, 180, 50]
    });
  }
}

// ==========================================
// 弾
// ==========================================

function updateBullets() {
  for (let i = bullets.length - 1; i >= 0; i--) {
    let b = bullets[i];

    b.x += b.vx;
    b.y += b.vy;
    b.life--;

    stroke(255, 230, 120);
    strokeWeight(3);

    line(
      b.x,
      b.y,
      b.x - b.vx * 1.5,
      b.y - b.vy * 1.5
    );

    // ゾンビとの当たり判定
    for (
      let j = zombies.length - 1;
      j >= 0;
      j--
    ) {
      let z = zombies[j];

      if (
        dist(b.x, b.y, z.x, z.y) <
        z.size * 0.5
      ) {
        z.hp--;

        z.hitFlash = 5;

        createBlood(
          z.x,
          z.y
        );

        bullets.splice(i, 1);

        if (z.hp <= 0) {
          score += z.size > 50 ? 50 : 20;

          createBigBlood(
            z.x,
            z.y
          );

          bloodSplats.push({
            x: z.x,
            y: z.y,
            size: random(30, 70)
          });

          zombies.splice(j, 1);
        }

        break;
      }
    }

    if (
      i < bullets.length &&
      (
        b.life <= 0 ||
        b.x < -50 ||
        b.x > width + 50 ||
        b.y < -50 ||
        b.y > height + 50
      )
    ) {
      bullets.splice(i, 1);
    }
  }
}

// ==========================================
// 血のエフェクト
// ==========================================

function createBlood(x, y) {
  for (let i = 0; i < 12; i++) {
    particles.push({
      x: x,
      y: y,

      vx: random(-4, 4),
      vy: random(-4, 4),

      life: random(20, 45),

      color: [170, 10, 10]
    });
  }
}

function createBigBlood(x, y) {
  for (let i = 0; i < 35; i++) {
    particles.push({
      x: x,
      y: y,

      vx: random(-7, 7),
      vy: random(-7, 7),

      life: random(30, 70),

      color: [180, 5, 5]
    });
  }
}

// ==========================================
// パーティクル
// ==========================================

function updateParticles() {
  noStroke();

  for (
    let i = particles.length - 1;
    i >= 0;
    i--
  ) {
    let p = particles[i];

    p.x += p.vx;
    p.y += p.vy;

    p.vx *= 0.96;
    p.vy *= 0.96;

    p.life--;

    fill(
      p.color[0],
      p.color[1],
      p.color[2],
      map(
        p.life,
        0,
        70,
        0,
        220
      )
    );

    circle(
      p.x,
      p.y,
      random(2, 6)
    );

    if (p.life <= 0) {
      particles.splice(i, 1);
    }
  }
}

// ==========================================
// 薬莢
// ==========================================

function updateCasings() {
  for (
    let i = casings.length - 1;
    i >= 0;
    i--
  ) {
    let c = casings[i];

    c.x += c.vx;
    c.y += c.vy;

    c.vx *= 0.97;
    c.vy *= 0.97;

    c.life--;

    fill(190, 150, 50);
    noStroke();

    ellipse(
      c.x,
      c.y,
      5,
      8
    );

    if (c.life <= 0) {
      casings.splice(i, 1);
    }
  }
}

function drawCrosshair() {
  stroke(220, 220, 220, 220);
  strokeWeight(2);
  noFill();

  circle(
    mouseX,
    mouseY,
    24
  );

  line(
    mouseX - 18,
    mouseY,
    mouseX - 7,
    mouseY
  );

  line(
    mouseX + 7,
    mouseY,
    mouseX + 18,
    mouseY
  );

  line(
    mouseX,
    mouseY - 18,
    mouseX,
    mouseY - 7
  );

  line(
    mouseX,
    mouseY + 7,
    mouseX,
    mouseY + 18
  );

  // マズルフラッシュ
  if (muzzleFlash > 0) {
    noStroke();

    fill(255, 200, 50, 150);

    circle(
      width / 2,
      height / 2,
      random(50, 80)
    );
  }
}

// ==========================================
// HUD
// ==========================================

function drawHUD() {
  // 画面を少し暗くする
  noStroke();

  for (let i = 0; i < 15; i++) {
    fill(0, 0, 0, 8);

    rect(
      i * 5,
      i * 5,
      width - i * 10,
      height - i * 10
    );
  }

  // HP
  fill(220);
  textSize(18);

  text(
    "HEALTH",
    25,
    height - 65
  );

  fill(50);

  rect(
    25,
    height - 52,
    220,
    18
  );

  if (hp > 50) {
    fill(40, 180, 70);
  } else if (hp > 25) {
    fill(220, 170, 30);
  } else {
    fill(200, 30, 30);
  }

  rect(
    25,
    height - 52,
    220 * (hp / 100),
    18
  );

  // 弾数
  textSize(35);

  if (ammo <= 3) {
    fill(220, 50, 50);
  } else {
    fill(230);
  }

  text(
    ammo + " / " + maxAmmo,
    width - 150,
    height - 40
  );

  // スコア
  fill(230);
  textSize(20);

  text(
    "SCORE  " + score,
    25,
    35
  );

  // ゾンビ数
  fill(150);
  textSize(14);

  text(
    "HOSTILES: " + zombies.length,
    25,
    55
  );

  // リロード
  if (reloading) {
    textAlign(CENTER, CENTER);

    fill(255);
    textSize(25);

    text(
      "RELOADING...",
      width / 2,
      height - 70
    );

    textAlign(LEFT, BASELINE);
  }
}

function drawGameOver() {
  background(5, 5, 5);

  for (let i = 0; i < 30; i++) {
    fill(100, 0, 0, 8);
    noStroke();

    ellipse(
      random(width),
      random(height),
      random(100, 500),
      random(100, 300)
    );
  }

  textAlign(CENTER, CENTER);

  fill(190, 25, 25);
  textSize(65);

  text(
    "YOU DIED",
    width / 2,
    height / 2 - 70
  );

  fill(230);
  textSize(28);

  text(
    "SCORE  " + score,
    width / 2,
    height / 2
  );

  fill(150);
  textSize(18);

  text(
    "CLICK TO RESTART",
    width / 2,
    height / 2 + 55
  );

  textAlign(LEFT, BASELINE);
}

// ==========================================
// 操作
// ==========================================

function mousePressed() {
  if (!gameStarted) {
    gameStarted = true;
    return;
  }

  if (gameOver) {
    restart();
    return;
  }

  shoot();
}

function keyPressed() {
  if (key === "r" || key === "R") {
    reload();
  }
}

function reload() {
  if (
    reloading ||
    ammo === maxAmmo ||
    gameOver
  ) {
    return;
  }

  reloading = true;

  // リロード時間
  reloadTimer = 10;
}

// ==========================================
// リスタート
// ==========================================

function restart() {
  zombies = [];
  bullets = [];
  particles = [];
  casings = [];
  bloodSplats = [];

  score = 0;
  hp = 100;
  ammo = maxAmmo;

  difficulty = 1;
  spawnTimer = 30;

  reloading = false;
  gameOver = false;
}
