// ============================================================
// 3D FPS MAZE (Improved Version - Spawn Facing Corridor)
// p5.js
// ============================================================

// ============================================================
// MAZE DEFINITION
// P: Player Start, G: Goal, #: Wall, ' ': Road
// ============================================================
const maze = [
  "###############",
  "#P            #",
  "# ##### ##### #",
  "#     #       #",
  "##### # #######",
  "#     #       #",
  "# ### ##### # #",
  "# #         # #",
  "# # ####### # #",
  "# #       #  G#",
  "# ####### #####",
  "#             #",
  "###############"
];

// ============================================================
// SETTINGS
// ============================================================
const CELL_SIZE = 80;
const WALL_HEIGHT = 120;
const PLAYER_HEIGHT = 45;

// カメラめり込み防止用半径
const PLAYER_RADIUS = 25;

const MOVE_SPEED = 240; // ピクセル/秒 (deltaTime用)
const MOUSE_SENSITIVITY = 0.0025;

// ============================================================
// STATE
// ============================================================
let playerX = 0;
let playerZ = 0;
// 初期の視界を通路方向（東向き）へ向けるため HALF_PI に設定
let playerYaw = 0;    // initMaze 時に設定
let playerPitch = 0;  // 垂直視点

let mazeRows = 0;
let mazeCols = 0;
let offsetX = 0;
let offsetZ = 0;

let goalPos = null;
let isCleared = false;
let startTime = 0;
let clearTime = 0;

let walls = [];
let keys = { w: false, a: false, s: false, d: false };

// ============================================================
// SETUP
// ============================================================
function setup() {
  createCanvas(windowWidth, windowHeight, WEBGL);
  noStroke();

  updatePerspective();
  initMaze();
  startTime = millis();

  window.addEventListener("keydown", handleKeyDown);
  window.addEventListener("keyup", handleKeyUp);
  window.addEventListener("blur", releaseAllKeys);
}

function updatePerspective() {
  const fov = PI / 3;
  const cameraZ = (height / 2.0) / tan(fov / 2.0);
  perspective(fov, width / height, 1, cameraZ * 10);
}

// ============================================================
// INITIALIZE MAZE
// ============================================================
function initMaze() {
  mazeRows = maze.length;
  mazeCols = maze[0].length;
  offsetX = (mazeCols * CELL_SIZE) / 2;
  offsetZ = (mazeRows * CELL_SIZE) / 2;

  walls = [];

  for (let r = 0; r < mazeRows; r++) {
    for (let c = 0; c < mazeCols; c++) {
      const tile = maze[r][c];
      const worldX = c * CELL_SIZE - offsetX + CELL_SIZE / 2;
      const worldZ = r * CELL_SIZE - offsetZ + CELL_SIZE / 2;

      if (tile === "#") {
        walls.push({ x: worldX, z: worldZ });
      } else if (tile === "P") {
        playerX = worldX;
        playerZ = worldZ;
        // 初期視界を通路が開いている方向（東＝右向き）へ向ける
        playerYaw = HALF_PI;
      } else if (tile === "G") {
        goalPos = { x: worldX, z: worldZ };
      }
    }
  }
}

// ============================================================
// INPUT EVENTS
// ============================================================
function handleKeyDown(event) {
  if (event.code === "KeyW") keys.w = true;
  if (event.code === "KeyA") keys.a = true;
  if (event.code === "KeyS") keys.s = true;
  if (event.code === "KeyD") keys.d = true;
}

function handleKeyUp(event) {
  if (event.code === "KeyW") keys.w = false;
  if (event.code === "KeyA") keys.a = false;
  if (event.code === "KeyS") keys.s = false;
  if (event.code === "KeyD") keys.d = false;
}

function releaseAllKeys() {
  keys.w = false;
  keys.a = false;
  keys.s = false;
  keys.d = false;
}

function mousePressed() {
  if (!document.pointerLockElement) {
    requestPointerLock();
  }
}

function mouseMoved() {
  if (document.pointerLockElement) {
    playerYaw += movedX * MOUSE_SENSITIVITY;
    playerPitch -= movedY * MOUSE_SENSITIVITY;
    playerPitch = constrain(playerPitch, -HALF_PI + 0.01, HALF_PI - 0.01);
  }
}

document.addEventListener("pointerlockchange", function () {
  if (!document.pointerLockElement) {
    releaseAllKeys();
  }
});

function windowResized() {
  resizeCanvas(windowWidth, windowHeight);
  updatePerspective();
}

// ============================================================
// MAIN LOOP
// ============================================================
function draw() {
  background(20, 24, 30);

  const dt = min(deltaTime / 1000, 0.1);

  if (!isCleared) {
    updatePlayer(dt);
    checkGoal();
  }

  updateCamera();

  ambientLight(120);
  directionalLight(255, 255, 255, -0.5, -1, -0.5);

  drawFloor();
  drawWalls();
  drawGoal();
}

// ============================================================
// PHYSICS & PLAYER
// ============================================================
function updatePlayer(dt) {
  let forward = 0;
  let strafe = 0;

  if (keys.w) forward += 1;
  if (keys.s) forward -= 1;
  if (keys.a) strafe -= 1;
  if (keys.d) strafe += 1;

  if (forward === 0 && strafe === 0) return;

  let dx = sin(playerYaw) * forward + cos(playerYaw) * strafe;
  let dz = -cos(playerYaw) * forward + sin(playerYaw) * strafe;

  const len = sqrt(dx * dx + dz * dz);
  if (len === 0) return;

  dx = (dx / len) * MOVE_SPEED * dt;
  dz = (dz / len) * MOVE_SPEED * dt;

  const nextX = playerX + dx;
  if (!checkGridCollision(nextX, playerZ)) {
    playerX = nextX;
  }

  const nextZ = playerZ + dz;
  if (!checkGridCollision(playerX, nextZ)) {
    playerZ = nextZ;
  }
}

function checkGridCollision(targetX, targetZ) {
  const centerCol = floor((targetX + offsetX) / CELL_SIZE);
  const centerRow = floor((targetZ + offsetZ) / CELL_SIZE);

  for (let r = centerRow - 1; r <= centerRow + 1; r++) {
    for (let c = centerCol - 1; c <= centerCol + 1; c++) {
      if (r < 0 || r >= mazeRows || c < 0 || c >= mazeCols) continue;

      if (maze[r][c] === "#") {
        const wallX = c * CELL_SIZE - offsetX + CELL_SIZE / 2;
        const wallZ = r * CELL_SIZE - offsetZ + CELL_SIZE / 2;

        const closestX = constrain(targetX, wallX - CELL_SIZE / 2, wallX + CELL_SIZE / 2);
        const closestZ = constrain(targetZ, wallZ - CELL_SIZE / 2, wallZ + CELL_SIZE / 2);

        if (dist(targetX, targetZ, closestX, closestZ) < PLAYER_RADIUS) {
          return true;
        }
      }
    }
  }
  return false;
}

function checkGoal() {
  if (!goalPos) return;
  if (dist(playerX, playerZ, goalPos.x, goalPos.z) < CELL_SIZE * 0.4) {
    isCleared = true;
    clearTime = ((millis() - startTime) / 1000).toFixed(2);
    if (document.exitPointerLock) {
      document.exitPointerLock();
    }
  }
}

// ============================================================
// CAMERA
// ============================================================
function updateCamera() {
  const lookDist = 100;
  const lookX = playerX + sin(playerYaw) * cos(playerPitch) * lookDist;
  const lookY = -PLAYER_HEIGHT + sin(playerPitch) * lookDist;
  const lookZ = playerZ - cos(playerYaw) * cos(playerPitch) * lookDist;

  camera(
    playerX, -PLAYER_HEIGHT, playerZ,
    lookX, lookY, lookZ,
    0, 1, 0
  );
}

// ============================================================
// RENDERING
// ============================================================
function drawFloor() {
  push();
  translate(0, 0, 0);
  rotateX(HALF_PI);
  ambientMaterial(40, 45, 50);
  plane(mazeCols * CELL_SIZE * 2, mazeRows * CELL_SIZE * 2);
  pop();
}

function drawWalls() {
  for (const wall of walls) {
    push();
    translate(wall.x, -WALL_HEIGHT / 2, wall.z);
    ambientMaterial(110, 120, 135);
    box(CELL_SIZE, WALL_HEIGHT, CELL_SIZE);
    pop();
  }
}

function drawGoal() {
  if (!goalPos) return;

  push();
  const floatY = -PLAYER_HEIGHT + sin(millis() * 0.005) * 8;
  translate(goalPos.x, floatY, goalPos.z);
  rotateY(millis() * 0.002);
  ambientMaterial(255, 215, 0);
  emissiveMaterial(80, 70, 0);
  box(24);
  pop();
}
