const COLORS = {
  bg1:[10,14,26], bg2:[20,26,48],
  panel2:'#1c2540',
  gold:'#f2c14e', goldDim:'#8a7038',
  ally1:'#f2c14e', ally2:'#7ec8e3', ally3:'#8b7ee8',
  enemy:'#c85a6e', elite:'#e0463c',
  hp:'#3ddc84', hpLow:'#e0463c',
  sp:'#5ac8fa', energy:'#f2c14e'
};

const ACT_THRESHOLD = 10000;

let party, enemies, sp, maxSp, log, phase, currentActor, pendingAction;
let winner, buttons, W = 900, H = 612;
let floatingTexts = [];
let speeches = [];
let shakeTimer = 0;

function makeParty(){
  return [
    {id:'tb', name:'開拓者', role:'散射', maxHp:1250, hp:1250, atk:118, spd:101,
     energy:20, maxEnergy:100, shield:0, alive:true, isParty:true, progress:0,
     col:COLORS.ally1, letter:'開'},
    {id:'m7', name:'三月なのか', role:'防護', maxHp:1050, hp:1050, atk:78, spd:104,
     energy:20, maxEnergy:100, shield:0, alive:true, isParty:true, progress:0,
     col:COLORS.ally2, letter:'三'},
    {id:'dh', name:'丹恒', role:'撃破', maxHp:980, hp:980, atk:146, spd:109,
     energy:20, maxEnergy:100, shield:0, alive:true, isParty:true, progress:0,
     col:COLORS.ally3, letter:'丹'}
  ];
}

function makeEnemies(){
  let list = [];
  for(let i=0;i<4;i++){
    list.push({id:'vr'+i, name:'ヴォイドレンジャー', maxHp:320, hp:320, atk:58, spd:92+i*2,
      alive:true, isParty:false, progress:0, elite:false, slot:i});
  }
  let elite = {id:'elite', name:'ヴォイド精鋭', maxHp:2100, hp:2100, atk:125, spd:88,
      alive:true, isParty:false, progress:0, elite:true, slot:4};
  list.splice(2,0,elite); // 5体構成: ヴォイドレンジャー×4 + 中央に精鋭
  return list;
}

function resetGame(){
  party = makeParty();
  enemies = makeEnemies();
  sp = 3; maxSp = 5;
  log = ['戦闘開始!'];
  phase = 'idle';
  currentActor = null;
  pendingAction = null;
  winner = null;
  floatingTexts = [];
  speeches = [];
}

function setup(){
  createCanvas(W,H);
  textFont("sans-serif");
  resetGame();
  setTimeout(nextTurn, 500);
}

// ---------- fraction helpers ----------
function X(f){ return f*W; }
function Y(f){ return f*H; }

// =========================================================
// TURN ORDER (Action Value system)
// =========================================================
function allUnits(){ return [...party, ...enemies]; }

function pickNextActor(){
  let units = allUnits().filter(u=>u.alive);
  if(units.length===0) return null;
  let dt = Math.min(...units.map(u=>(ACT_THRESHOLD-u.progress)/u.spd));
  units.forEach(u=> u.progress += u.spd*dt);
  let ready = units.filter(u=>u.progress >= ACT_THRESHOLD - 0.01);
  ready.sort((a,b)=> b.spd - a.spd);
  let actor = ready[0];
  actor.progress -= ACT_THRESHOLD;
  return actor;
}

function simulateOrder(n){
  let clones = allUnits().filter(u=>u.alive).map(u=>({id:u.id, spd:u.spd, progress:u.progress, isParty:u.isParty}));
  let result = [];
  for(let i=0;i<n;i++){
    if(clones.length===0) break;
    let dt = Math.min(...clones.map(u=>(ACT_THRESHOLD-u.progress)/u.spd));
    clones.forEach(u=> u.progress += u.spd*dt);
    let ready = clones.filter(u=>u.progress>=ACT_THRESHOLD-0.01);
    ready.sort((a,b)=>b.spd-a.spd);
    let actor = ready[0];
    actor.progress -= ACT_THRESHOLD;
    result.push(actor);
  }
  return result;
}

// =========================================================
// GAME FLOW
// =========================================================
function nextTurn(){
  if(winner) return;
  let actor = pickNextActor();
  if(!actor){ return; }
  currentActor = actor;
  if(actor.isParty){
    phase = 'selectAction';
    pushLog(`${actor.name} の行動`);
  } else {
    phase = 'enemyActing';
    setTimeout(()=>enemyAct(actor), 700);
  }
}

function pushLog(msg){
  log.push(msg);
  if(log.length>6) log.shift();
}

function gainSp(n){ sp = constrain(sp+n, 0, maxSp); }
function spendSp(n){ sp = constrain(sp-n, 0, maxSp); }
function gainEnergy(u,n){ u.energy = constrain(u.energy+n, 0, u.maxEnergy); }

function addFloat(target, text, color){
  floatingTexts.push({target, text, color, t:0});
}

// ---------- セリフ(オリジナル、ゲーム本編の台本とは無関係) ----------
const QUOTES = {
  tb: {
    attack: ['行くぞ!', 'これで決める!', 'まだまだ!'],
    skill: ['散らばれ!', 'まとめて片付ける!'],
    ult_single: ['ここで終わらせる!'],
    ult_aoe: ['全部纏めて叩く!']
  },
  m7: {
    attack: ['えいっ!', '当たれー!'],
    skill: ['大丈夫、任せて!'],
    ult: ['反撃の時間だよ!']
  },
  dh: {
    attack: ['……', '退け。'],
    skill: ['そこまでだ。'],
    ult: ['この一閃で終わる。']
  },
  voidRanger: ['……ノイズ……', 'シュルル……'],
  elite: ['虚無に還れ……', '崩れよ……']
};

function randomFrom(arr){ return arr[Math.floor(Math.random()*arr.length)]; }

function showSpeech(unit, text){
  speeches = speeches.filter(s=>s.unit!==unit);
  speeches.push({unit, text, t:0});
}

function dealDamage(target, dmg){
  dmg = Math.round(dmg);
  let absorbed = 0;
  if(target.shield>0){
    absorbed = Math.min(target.shield, dmg);
    target.shield -= absorbed;
    dmg -= absorbed;
  }
  target.hp = Math.max(0, target.hp - dmg);
  addFloat(target, absorbed>0 && dmg===0 ? `-${absorbed}(シールド)` : `-${dmg}`, '#ff8a80');
  if(target.isParty && dmg>0){
    gainEnergy(target, 10);
  }
  if(target.hp<=0 && target.alive){
    target.alive = false;
    pushLog(`${target.name} は戦闘不能になった`);
  }
  shakeTimer = 8;
  return dmg;
}

function checkWinLose(){
  if(enemies.every(e=>!e.alive)){ winner='party'; phase='gameover'; pushLog('勝利!'); return true; }
  if(party.every(p=>!p.alive)){ winner='enemy'; phase='gameover'; pushLog('敗北...'); return true; }
  return false;
}

function endActionAndProceed(){
  if(checkWinLose()) return;
  phase = 'idle';
  currentActor = null;
  pendingAction = null;
  setTimeout(nextTurn, 650);
}

// ---------- Player action execution ----------
function performAction(actor, type, opts={}){
  const target = opts.target;

  if(actor.id==='tb'){
    if(type==='attack'){
      dealDamage(target, actor.atk*1.0);
      gainSp(1); gainEnergy(actor,20);
      pushLog(`${actor.name} の通常攻撃 → ${target.name}`);
      showSpeech(actor, randomFrom(QUOTES.tb.attack));
    } else if(type==='skill'){
      spendSp(1);
      enemies.filter(e=>e.alive).forEach(e=> dealDamage(e, actor.atk*0.5));
      gainEnergy(actor,30);
      pushLog(`${actor.name} のスキル『拡散攻撃』(敵全体にダメージ)`);
      showSpeech(actor, randomFrom(QUOTES.tb.skill));
    } else if(type==='ult'){
      if(opts.mode==='aoe'){
        // 拡散: 対象+左右隣接の敵を攻撃
        let idx = enemies.indexOf(target);
        dealDamage(target, actor.atk*2.0);
        [idx-1, idx+1].forEach(i=>{
          let side = enemies[i];
          if(side && side.alive) dealDamage(side, actor.atk*1.0);
        });
        pushLog(`${actor.name} の必殺技(拡散:対象+隣接) → ${target.name}`);
        showSpeech(actor, randomFrom(QUOTES.tb.ult_aoe));
      } else {
        dealDamage(target, actor.atk*2.8);
        pushLog(`${actor.name} の必殺技(単体) → ${target.name}`);
        showSpeech(actor, randomFrom(QUOTES.tb.ult_single));
      }
      actor.energy = 0;
    }

  } else if(actor.id==='m7'){
    if(type==='attack'){
      dealDamage(target, actor.atk*1.0);
      gainSp(1); gainEnergy(actor,20);
      pushLog(`${actor.name} の通常攻撃 → ${target.name}`);
      showSpeech(actor, randomFrom(QUOTES.m7.attack));
    } else if(type==='skill'){
      // スキル: 味方単体にバリア
      spendSp(1);
      target.shield = Math.min(600, target.shield+280);
      gainEnergy(actor,30);
      pushLog(`${actor.name} のスキル『護りの盾』 → ${target.name}`);
      showSpeech(actor, randomFrom(QUOTES.m7.skill));
    } else if(type==='ult'){
      // 必殺技: 敵全体攻撃+行動順ダウン
      enemies.filter(e=>e.alive).forEach(e=>{
        dealDamage(e, actor.atk*0.9);
        e.progress = Math.max(0, e.progress - ACT_THRESHOLD*0.35);
      });
      actor.energy = 0;
      pushLog(`${actor.name} の必殺技(全体攻撃+行動順ダウン)`);
      showSpeech(actor, randomFrom(QUOTES.m7.ult));
    }

  } else if(actor.id==='dh'){
    if(type==='attack'){
      dealDamage(target, actor.atk*1.0);
      gainSp(1); gainEnergy(actor,20);
      pushLog(`${actor.name} の通常攻撃 → ${target.name}`);
      showSpeech(actor, randomFrom(QUOTES.dh.attack));
    } else if(type==='skill'){
      spendSp(1);
      dealDamage(target, actor.atk*1.8);
      gainEnergy(actor,30);
      pushLog(`${actor.name} のスキル『強襲』 → ${target.name}`);
      showSpeech(actor, randomFrom(QUOTES.dh.skill));
    } else if(type==='ult'){
      dealDamage(target, actor.atk*3.5);
      actor.energy = 0;
      pushLog(`${actor.name} の必殺技(単体大破壊) → ${target.name}`);
      showSpeech(actor, randomFrom(QUOTES.dh.ult));
    }
  }

  endActionAndProceed();
}

// ---------- Enemy AI ----------
function enemyAct(actor){
  let alivePartyMembers = party.filter(p=>p.alive);
  if(alivePartyMembers.length===0){ checkWinLose(); return; }
  if(actor.elite && Math.random()<0.35){
    alivePartyMembers.forEach(p=> dealDamage(p, actor.atk*0.55));
    pushLog(`${actor.name} の全体攻撃!`);
    showSpeech(actor, randomFrom(QUOTES.elite));
  } else {
    alivePartyMembers.sort((a,b)=> (a.hp/a.maxHp) - (b.hp/b.maxHp));
    let target = alivePartyMembers[0];
    dealDamage(target, actor.atk*(actor.elite?1.15:1.0));
    pushLog(`${actor.name} の攻撃 → ${target.name}`);
    showSpeech(actor, randomFrom(actor.elite ? QUOTES.elite : QUOTES.voidRanger));
  }
  endActionAndProceed();
}

// =========================================================
// LAYOUT
// =========================================================
function enemySlotX(i){
  const xs = [0.12,0.31,0.5,0.69,0.88]; // vr0,vr1,elite,vr2,vr3 (5体構成)
  return xs[i];
}

function getEnemyPos(e, idx){
  let fx = enemySlotX(idx);
  let fy = e.elite ? 0.40 : 0.32;
  return {x:X(fx), y:Y(fy), r: e.elite ? X(0.052) : X(0.032)};
}

function getPartyPos(idx){
  const xs=[0.20,0.5,0.80];
  return {x:X(xs[idx]), y:Y(0.63), r:X(0.045)};
}

// =========================================================
// DRAW
// =========================================================
function draw(){
  let dx=0, dy=0;
  if(shakeTimer>0){ dx=random(-3,3); dy=random(-3,3); shakeTimer--; }
  push();
  translate(dx,dy);
  drawBackground();
  drawTurnOrderBar();
  drawLogPanel();
  drawEnemies();
  drawParty();
  drawSpRow();
  drawFloatingTexts();
  drawSpeeches();
  drawActionUI();
  pop();
  if(phase==='gameover') drawGameOver();
}

function drawBackground(){
  noStroke();
  for(let i=0;i<H;i+=2){
    let t = i/H;
    let c = lerpColor(color(...COLORS.bg1), color(...COLORS.bg2), t);
    fill(c); rect(0,i,W,2);
  }
  randomSeed(42);
  fill(255,255,255,60);
  for(let i=0;i<60;i++){
    let sx = (i*97)%W, sy=(i*53)%H;
    ellipse(sx, sy*0.6+10, 1.6,1.6);
  }
}

function drawTurnOrderBar(){
  const order = [currentActor, ...simulateOrder(7)].filter(Boolean);
  const barY = Y(0.045);
  noStroke(); fill(15,20,35,180);
  rect(X(0.02), Y(0.01), X(0.62), Y(0.08), 8);
  let x0 = X(0.05);
  order.slice(0,7).forEach((u,i)=>{
    let full = allUnits().find(x=>x.id===u.id);
    if(!full || !full.alive) return;
    let x = x0 + i*X(0.075);
    let r = i===0 ? X(0.024) : X(0.018);
    fill(full.isParty ? full.col : (full.elite? COLORS.elite : COLORS.enemy));
    stroke(i===0? COLORS.gold : 'rgba(255,255,255,0.25)');
    strokeWeight(i===0?2:1);
    ellipse(x, barY, r*2, r*2);
    noStroke();
    textAlign(CENTER,CENTER); textSize(X(0.014));
    fill('#0a0e1a');
    text(full.isParty? full.letter : (full.elite?'精':'虚'), x, barY+1);
  });
}

function drawLogPanel(){
  noStroke(); fill(15,20,35,150);
  rect(X(0.02), Y(0.115), X(0.4), Y(0.14), 8);
  textAlign(LEFT, TOP); textSize(X(0.0135));
  let lines = log.slice(-5);
  lines.forEach((l,i)=>{
    fill(i===lines.length-1? '#eef2f8' : '#77839f');
    text(l, X(0.03), Y(0.125)+i*Y(0.026));
  });
}

function drawEnemies(){
  enemies.forEach((e,idx)=>{
    if(!e.alive) return;
    let {x,y,r} = getEnemyPos(e, idx);
    let selectable = phase==='selectTarget' && (!pendingAction || pendingAction.targetType!=='ally') && e.alive;
    let hover = selectable && dist(mouseX, mouseY, x, y) < r+6;

    noStroke();
    fill(e.elite ? COLORS.elite : COLORS.enemy);
    if(hover){ stroke(COLORS.gold); strokeWeight(3); } else { noStroke(); }
    ellipse(x,y,r*2,r*2);

    noStroke(); fill(255,255,255,230);
    textAlign(CENTER,CENTER); textSize(r*0.7);
    text(e.elite? '精' : '虚', x, y+1);

    textAlign(CENTER,CENTER);
    fill('#eef2f8'); textSize(X(0.012));
    text(e.name, x, y-r-X(0.022));
    drawBar(x-r, y-r-X(0.012), r*2, X(0.008), e.hp/e.maxHp, COLORS.hp, COLORS.hpLow);

    if(selectable){
      fill(COLORS.gold); textSize(X(0.011));
      text('▼ 選択', x, y+r+X(0.018));
    }
  });
}

function drawParty(){
  party.forEach((p,idx)=>{
    let {x,y,r} = getPartyPos(idx);
    let isCurrent = currentActor && currentActor.id===p.id;
    let selectableAlly = phase==='selectTarget' && pendingAction && pendingAction.targetType==='ally' && p.alive;
    let hoverAlly = selectableAlly && dist(mouseX,mouseY,x,y) < r+10;

    noStroke();
    fill(p.alive ? p.col : '#333a4d');
    if(hoverAlly){ stroke(COLORS.gold); strokeWeight(4); }
    else if(isCurrent){ stroke(COLORS.gold); strokeWeight(3); }
    else { noStroke(); }
    ellipse(x,y,r*2,r*2);

    noStroke(); fill('#0a0e1a'); textAlign(CENTER,CENTER); textSize(r*0.6);
    text(p.letter, x, y+1);

    fill(p.alive? '#eef2f8':'#666f85'); textSize(X(0.013));
    text(p.name, x, y-r-X(0.022));
    fill('#9aa5c0'); textSize(X(0.010));
    text(p.role, x, y+r+X(0.02));

    drawBar(x-r, y+r+X(0.038), r*2, X(0.009), Math.max(0,p.hp)/p.maxHp, COLORS.hp, COLORS.hpLow);
    textAlign(CENTER,CENTER); fill('#eef2f8'); textSize(X(0.0105));
    text(`${Math.max(0,Math.round(p.hp))}/${p.maxHp}`, x, y+r+X(0.055));

    if(p.shield>0){
      fill(COLORS.sp); textSize(X(0.010));
      text(`シールド ${Math.round(p.shield)}`, x, y+r+X(0.075));
    }

    drawBar(x-r, y+r+X(0.09), r*2, X(0.007), p.energy/p.maxEnergy, COLORS.energy, COLORS.energy);
    fill('#9aa5c0'); textSize(X(0.0095));
    text(`必殺技 ${Math.round(p.energy)}/100`, x, y+r+X(0.105));

    if(selectableAlly){
      fill(COLORS.gold); textSize(X(0.011));
      text('▼ 選択', x, y+r+X(0.125));
    }
  });
}

function drawBar(x,y,w,h,pct,colGood,colBad){
  pct = constrain(pct,0,1);
  noStroke(); fill(0,0,0,120); rect(x,y,w,h,3);
  fill(pct<0.3 ? colBad : colGood); rect(x,y,w*pct,h,3);
  noFill(); stroke(255,255,255,40); strokeWeight(1); rect(x,y,w,h,3); noStroke();
}

function drawSpRow(){
  const cx = X(0.5), cy = Y(0.80);
  const n = maxSp, size = X(0.018), gap = X(0.026);
  textAlign(CENTER,CENTER); fill('#9aa5c0'); textSize(X(0.012));
  text('SP', cx, cy-X(0.03));
  for(let i=0;i<n;i++){
    let x = cx - (n-1)*gap/2 + i*gap;
    push(); translate(x,cy); rotate(PI/4);
    noStroke(); fill(i<sp ? COLORS.sp : 'rgba(255,255,255,0.12)');
    rectMode(CENTER); rect(0,0,size,size,2);
    pop();
  }
}

function drawFloatingTexts(){
  floatingTexts.forEach(ft=>{
    if(!ft.target) return;
    let pos;
    let pIdx = party.indexOf(ft.target);
    if(pIdx>=0) pos = getPartyPos(pIdx);
    else {
      let eIdx = enemies.indexOf(ft.target);
      if(eIdx>=0) pos = getEnemyPos(ft.target, eIdx);
    }
    if(!pos) return;
    noStroke(); fill(ft.color); textAlign(CENTER,CENTER); textSize(X(0.016));
    text(ft.text, pos.x, pos.y - pos.r - X(0.05) - ft.t*0.4);
    ft.t += 1;
  });
  floatingTexts = floatingTexts.filter(f=>f.t<40);
}

function drawSpeeches(){
  speeches.forEach(s=>{
    let pos;
    let pIdx = party.indexOf(s.unit);
    if(pIdx>=0) pos = getPartyPos(pIdx);
    else {
      let eIdx = enemies.indexOf(s.unit);
      if(eIdx>=0) pos = getEnemyPos(s.unit, eIdx);
    }
    if(!pos) return;

    textSize(X(0.0135));
    let tw = textWidth(s.text);
    let padX = X(0.014), bw = tw + padX*2, bh = X(0.032);
    let bx = pos.x, by = pos.y - pos.r - X(0.075);

    noStroke(); fill(255,255,255,235);
    rectMode(CENTER);
    rect(bx, by, bw, bh, 9);
    triangle(bx-X(0.008), by+bh/2-1, bx+X(0.008), by+bh/2-1, bx, by+bh/2+X(0.012));
    rectMode(CORNER);

    fill('#141b2e'); textAlign(CENTER,CENTER);
    text(s.text, bx, by);

    s.t += 1;
  });
  speeches = speeches.filter(s=>s.t<95);
}

// =========================================================
// ACTION UI (buttons)
// =========================================================
function drawActionUI(){
  buttons = [];
  const barY = Y(0.90);
  noStroke(); fill(15,20,35,200);
  rect(0, Y(0.885), W, Y(0.115));

  if(phase==='selectAction' && currentActor){
    const a = currentActor;
    const labels = [
      {key:'attack', label:'通常攻撃', enabled:true},
      {key:'skill', label:'スキル', enabled: sp>=1},
      {key:'ult', label:'必殺技', enabled: a.energy>=a.maxEnergy}
    ];
    const bw = X(0.16), bh = Y(0.06), gap=X(0.02);
    let totalW = bw*3+gap*2;
    let x0 = X(0.5)-totalW/2;
    labels.forEach((b,i)=>{
      let x = x0+i*(bw+gap), y = barY - bh/2;
      drawButton(x,y,bw,bh,b.label,b.enabled);
      buttons.push({x,y,w:bw,h:bh,action:b.key,enabled:b.enabled});
    });
  } else if(phase==='selectUltMode' && currentActor){
    const bw=X(0.18), bh=Y(0.06), gap=X(0.03);
    let x0 = X(0.5)-(bw*2+gap)/2;
    drawButton(x0, barY-bh/2, bw,bh,'拡散(隣接)',true);
    buttons.push({x:x0,y:barY-bh/2,w:bw,h:bh,action:'ult_aoe',enabled:true});
    drawButton(x0+bw+gap, barY-bh/2, bw,bh,'単体選択',true);
    buttons.push({x:x0+bw+gap,y:barY-bh/2,w:bw,h:bh,action:'ult_single',enabled:true});
    drawCancelButton();
  } else if(phase==='selectTarget'){
    textAlign(CENTER,CENTER); fill(COLORS.gold); textSize(X(0.016));
    let msg = (pendingAction && pendingAction.targetType==='ally') ? '対象の味方をクリックしてください' : '攻撃対象の敵をクリックしてください';
    text(msg, X(0.5), barY);
    drawCancelButton();
  }
}

function drawCancelButton(){
  const bw=X(0.1), bh=Y(0.05);
  const x=W-bw-X(0.02), y=Y(0.885)-bh-Y(0.01);
  drawButton(x,y,bw,bh,'キャンセル',true, true);
  buttons.push({x,y,w:bw,h:bh,action:'cancel',enabled:true});
}

function drawButton(x,y,w,h,label,enabled,subtle){
  noStroke();
  fill(enabled ? (subtle? 'rgba(255,255,255,0.08)' : COLORS.panel2) : 'rgba(255,255,255,0.05)');
  stroke(enabled? COLORS.goldDim : 'rgba(255,255,255,0.1)');
  strokeWeight(1.5);
  rect(x,y,w,h,8);
  noStroke();
  fill(enabled ? '#eef2f8' : '#555d70');
  textAlign(CENTER,CENTER); textSize(min(w,h)*0.28);
  text(label, x+w/2, y+h/2);
}

function drawGameOver(){
  noStroke(); fill(0,0,0,190); rect(0,0,W,H);
  textAlign(CENTER,CENTER);
  fill(winner==='party' ? COLORS.gold : COLORS.enemy);
  textSize(X(0.05));
  text(winner==='party' ? '勝利' : '敗北', X(0.5), Y(0.42));
  fill('#eef2f8'); textSize(X(0.016));
  text(winner==='party' ? '敵を全て撃破した' : '味方が全滅した', X(0.5), Y(0.5));

  const bw=X(0.2), bh=Y(0.07);
  const x=X(0.5)-bw/2, y=Y(0.58);
  drawButton(x,y,bw,bh,'リスタート',true);
  buttons=[{x,y,w:bw,h:bh,action:'restart',enabled:true}];
}

function mousePressed(){
  const mx = mouseX, my = mouseY;

  if(buttons){
    for(let b of buttons){
      if(b.enabled && mx>=b.x && mx<=b.x+b.w && my>=b.y && my<=b.y+b.h){
        handleButton(b.action);
        return;
      }
    }
  }

  if(phase==='selectTarget'){
    if(pendingAction.targetType==='ally'){
      party.forEach((p,idx)=>{
        if(!p.alive) return;
        let {x,y,r} = getPartyPos(idx);
        if(dist(mx,my,x,y) < r+10){
          performAction(currentActor, pendingAction.type, {target:p, mode:pendingAction.mode});
        }
      });
    } else {
      enemies.forEach((e,idx)=>{
        if(!e.alive) return;
        let {x,y,r} = getEnemyPos(e, idx);
        if(dist(mx,my,x,y) < r+6){
          performAction(currentActor, pendingAction.type, {target:e, mode:pendingAction.mode});
        }
      });
    }
  }
}

function handleButton(action){
  if(action==='restart'){ resetGame(); setTimeout(nextTurn,500); return; }
  if(action==='cancel'){ phase='selectAction'; pendingAction=null; return; }

  if(phase==='selectAction'){
    if(action==='attack'){ pendingAction={type:'attack', targetType:'enemy'}; phase='selectTarget'; }
    else if(action==='skill'){
      if(currentActor.id==='tb'){
        performAction(currentActor,'skill',{});
      } else if(currentActor.id==='m7'){
        pendingAction={type:'skill', targetType:'ally'}; phase='selectTarget';
      } else {
        pendingAction={type:'skill', targetType:'enemy'}; phase='selectTarget';
      }
    } else if(action==='ult'){
      if(currentActor.id==='tb'){ phase='selectUltMode'; }
      else if(currentActor.id==='m7'){ performAction(currentActor,'ult',{}); }
      else { pendingAction={type:'ult', targetType:'enemy'}; phase='selectTarget'; }
    }
  } else if(phase==='selectUltMode'){
    if(action==='ult_aoe'){ pendingAction={type:'ult',mode:'aoe',targetType:'enemy'}; phase='selectTarget'; }
    else if(action==='ult_single'){ pendingAction={type:'ult',mode:'single',targetType:'enemy'}; phase='selectTarget'; }
  }
}
