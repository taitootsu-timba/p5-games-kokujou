let lives = 3;

let cameraX = 0;

let gameState = "play";


// ==========================================
// SETUP
// ==========================================

function setup() {

  createCanvas(800, 450);

  startLevel(1);
}


// ==========================================
// DRAW
// ==========================================

function draw() {

  drawBackground();

  if (gameState === "play") {

    updateGame();
  }

  drawWorld();

  drawUI();


  if (gameState === "levelClear") {

    drawLevelClear();
  }


  if (gameState === "gameover") {

    drawGameOver();
  }


  if (gameState === "allClear") {

    drawAllClear();
  }
}

  platforms = [];
  blocks = [];
  coins = [];

  cameraX = 0;

  createLevel(n);

  gameState = "play";
}

function makeGround(x, w) {

  platforms.push(
    new Platform(
      x,
      400,
      w,
      50
    )
  );
}


function makePlatform(x, y, w) {

  platforms.push(
    new Platform(
      x,
      y,
      w,
      30
    )
  );
}


function makeBlock(x, y) {

  blocks.push(
    new QuestionBlock(x, y)
  );
}


function makeEnemy(x, speed) {

  enemies.push(
    new Enemy(
      x,
      368,
      speed
    )
  );
}


function makeCoin(x, y) {

  coins.push({

    x: x,
    y: y,

    taken: false,

    angle: 0
  });
}


function addCoinsLine(
  start,
  end,
  y
) {

  for (
    let x = start;
    x <= end;
    x += 250
  ) {

    makeCoin(x, y);
  }
}


// ==========================================
// ゲーム更新
// ==========================================

function updateGame() {

  player.update();


  for (let enemy of enemies) {

    enemy.update();
  }


  for (let item of items) {

    item.update();
  }


  checkPlatforms();

  checkBlocks();

  checkCoins();

  checkItems();

  checkEnemies();


  // 落下
  if (player.y > 520) {

    damage();
  }


  // ゴール
  if (
    player.x >= player.goalX
  ) {

    gameState = "levelClear";
  }


  // カメラ
  cameraX =
    player.x - 250;


  cameraX =
    constrain(
      cameraX,
      0,
      player.goalX - 550
    );
}

class Player {

  constructor(x, y) {

    this.x = x;
    this.y = y;

    this.w = 32;
    this.h = 48;

    this.vx = 0;
    this.vy = 0;

    this.speed = 5;

    this.jumpPower = 12;

    this.grounded = false;

    this.big = false;

    this.invincible = 0;

    this.star = 0;

    this.goalX = 3000;
  }


  update() {

    // 左
    if (
      keyIsDown(LEFT_ARROW)
    ) {

      this.vx = -this.speed;
    }


    // 右
    else if (
      keyIsDown(RIGHT_ARROW)
    ) {

      this.vx = this.speed;
    }


    // 何も押していない
    else {

      this.vx *= 0.75;

      if (
        abs(this.vx) < 0.1
      ) {

        this.vx = 0;
      }
    }


    // 重力
    this.vy += 0.55;


    // 移動
    this.x += this.vx;

    this.y += this.vy;


    // 左端
    if (this.x < 0) {

      this.x = 0;
    }


    // 無敵
    if (
      this.invincible > 0
    ) {

      this.invincible--;
    }


    if (
      this.star > 0
    ) {

      this.star--;
    }
  }


  jump() {

    if (this.grounded) {

      this.vy =
        -this.jumpPower;

      this.grounded = false;
    }
  }


  draw() {

    push();


    translate(
      this.x - cameraX,
      this.y
    );


    // ダメージ中
    if (
      this.invincible > 0 &&
      frameCount % 6 < 3
    ) {

      pop();

      return;
    }


    // スター
    if (
      this.star > 0
    ) {

      fill(
        random(255),
        random(200, 255),
        0
      );

      circle(
        16,
        25,
        70
      );
    }


    let h =
      this.big
        ? 60
        : 48;


    // 帽子
    fill("#e52521");

    rect(
      5,
      0,
      25,
      10,
      5
    );


    // 顔
    fill("#ffc27d");

    rect(
      7,
      8,
      23,
      20,
      5
    );


    // 目
    fill("#222");

    rect(
      22,
      13,
      3,
      5
    );


    // 服
    fill("#2468d7");

    rect(
      5,
      28,
      25,
      h - 28,
      5
    );


    // 手
    fill("#ffc27d");

    rect(
      0,
      30,
      7,
      18,
      4
    );

    rect(
      30,
      30,
      7,
      18,
      4
    );


    // 靴
    fill("#713b1d");

    rect(
      3,
      h - 5,
      13,
      8,
      3
    );

    rect(
      20,
      h - 5,
      13,
      8,
      3
    );


    pop();
  }
}

    rect(
      this.x - cameraX,
      this.y,
      this.w,
      8
    );
  }
}


// ==========================================
// ？ブロック
// ==========================================

class QuestionBlock {

  constructor(x, y) {

    this.x = x;
    this.y = y;

    this.w = 45;
    this.h = 45;

    this.used = false;

    this.bump = 0;
  }


  draw() {

    if (
      this.bump > 0
    ) {

      this.bump--;
    }


    let y =
      this.y - this.bump;


    fill(
      this.used
        ? "#888"
        : "#e89b32"
    );


    stroke("#9b5b1b");

    rect(
      this.x - cameraX,
      y,
      this.w,
      this.h
    );


    noStroke();


    if (!this.used) {

      fill("#fff2a0");

      textAlign(
        CENTER,
        CENTER
      );

      textSize(28);

      text(
        "?",
        this.x -
        cameraX +
        22,
        y + 22
      );
    }
  }
}


// ==========================================
// 敵
// ==========================================

class Enemy {

  constructor(
    x,
    y,
    speed
  ) {

    this.x = x;
    this.y = y;

    this.w = 34;
    this.h = 32;

    this.vx = speed;

    this.dead = false;
  }


  update() {

    if (this.dead) {
      return;
    }


    this.x += this.vx;


    // 壁のように端で反転
    if (
      this.x < 0 ||
      this.x > 3900
    ) {

      this.vx *= -1;
    }
  }


  draw() {

    if (this.dead) {
      return;
    }


    let x =
      this.x - cameraX;


    fill("#8b451f");

    ellipse(
      x + 17,
      this.y + 17,
      34,
      30
    );


    fill("white");

    rect(
      x + 8,
      this.y + 8,
      7,
      8
    );

    rect(
      x + 20,
      this.y + 8,
      7,
      8
    );


    fill("#222");

    rect(
      x + 11,
      this.y + 10,
      3,
      4
    );

    rect(
      x + 23,
      this.y + 10,
      3,
      4
    );
  }
}


// ==========================================
// アイテム
// ==========================================

class Item {

  constructor(
    x,
    y,
    type
  ) {

    this.x = x;
    this.y = y;

    this.type = type;

    this.w = 30;
    this.h = 30;

    this.vx = 0;
    this.vy = -6;

    this.active = true;
  }


  update() {

    if (!this.active) {
      return;
    }


    this.vy += 0.4;

    this.x += this.vx;

    this.y += this.vy;


    for (
      let p of platforms
    ) {

      if (
        this.x + this.w > p.x &&
        this.x < p.x + p.w &&
        this.y + this.h >= p.y &&
        this.y + this.h <=
        p.y + 20 &&
        this.vy >= 0
      ) {

        this.y =
          p.y - this.h;


        if (
          this.type ===
          "mushroom"
        ) {

          this.vy = 0;

          if (
            this.vx === 0
          ) {

            this.vx = 2;
          }
        }


        if (
          this.type === "star"
        ) {

          this.vy = -8;

          if (
            this.vx === 0
          ) {

            this.vx = 2;
          }
        }
      }
    }
  }


  draw() {

    if (!this.active) {
      return;
    }


    let x =
      this.x - cameraX;


    // コイン
    if (
      this.type === "coin"
    ) {

      fill("#ffd21f");

      ellipse(
        x + 15,
        this.y + 15,
        25,
        30
      );
    }


    // キノコ
    if (
      this.type === "mushroom"
    ) {

      fill("#fff0c2");

      rect(
        x + 8,
        this.y + 15,
        15,
        15,
        4
      );


      fill("#ef3b3b");

      arc(
        x + 15,
        this.y + 17,
        32,
        30,
        PI,
        TWO_PI
      );


      fill("white");

      circle(
        x + 8,
        this.y + 13,
        8
      );

      circle(
        x + 22,
        this.y + 13,
        8
      );
    }


    // スター
    if (
      this.type === "star"
    ) {

      fill("#ffd21f");

      drawStar(
        x + 15,
        this.y + 15,
        17,
        8
      );
    }
  }
}


// ==========================================
// 星
// ==========================================

function drawStar(
  x,
  y,
  r1,
  r2
) {

  beginShape();

  for (
    let i = 0;
    i < 10;
    i++
  ) {

    let a =
      -PI / 2 +
      i * PI / 5;

    let r =
      i % 2 === 0
        ? r1
        : r2;

    vertex(
      x + cos(a) * r,
      y + sin(a) * r
    );
  }

  endShape(CLOSE);
}


// ==========================================
// 足場判定
// ==========================================

function checkPlatforms() {

  player.grounded = false;


  for (
    let p of platforms
  ) {

    if (
      player.vy >= 0 &&
      player.x + player.w > p.x &&
      player.x < p.x + p.w &&
      player.y + player.h >= p.y &&
      player.y + player.h <=
      p.y + 25
    ) {

      player.y =
        p.y - player.h;

      player.vy = 0;

      player.grounded = true;
    }
  }
}


// ==========================================
// ブロック判定
// ==========================================

function checkBlocks() {

  for (
    let b of blocks
  ) {

    if (b.used) {
      continue;
    }


    if (
      player.x + player.w > b.x &&
      player.x < b.x + b.w &&
      player.y < b.y + b.h &&
      player.y + player.h > b.y
    ) {


      // 下から叩く
      if (
        player.vy < 0 &&
        player.y > b.y
      ) {

        b.used = true;

        b.bump = 10;

        player.y =
          b.y + b.h;

        player.vy = 1;

        score += 50;

        createItem(b);
      }
    }
  }
}


// ==========================================
// アイテム
// ==========================================

function createItem(block) {

  let r = random();

  let type;


  if (r < 0.5) {

    type = "coin";

  } else if (r < 0.8) {

    type = "mushroom";

  } else {

    type = "star";
  }


  let item =
    new Item(
      block.x + 7,
      block.y - 30,
      type
    );


  if (
    type === "mushroom" ||
    type === "star"
  ) {

    item.vx = 2;
  }


  items.push(item);
}


// ==========================================
// コイン判定
// ==========================================

function checkCoins() {

  for (
    let c of coins
  ) {

    if (c.taken) {
      continue;
    }


    let d =
      dist(
        player.x + 16,
        player.y + 24,
        c.x,
        c.y
      );


    if (d < 35) {

      c.taken = true;

      score += 100;
    }
  }
}


// ==========================================
// アイテム取得
// ==========================================

function checkItems() {

  for (
    let item of items
  ) {

    if (!item.active) {
      continue;
    }


    if (
      player.x <
      item.x + item.w &&
      player.x + player.w >
      item.x &&
      player.y <
      item.y + item.h &&
      player.y + player.h >
      item.y
    ) {


      if (
        item.type === "coin"
      ) {

        score += 100;

        item.active = false;
      }


      if (
        item.type === "mushroom"
      ) {

        player.big = true;

        score += 500;

        item.active = false;
      }


      if (
        item.type === "star"
      ) {

        player.star = 600;

        score += 1000;

        item.active = false;
      }
    }
  }
}


// ==========================================
// 敵判定
// ==========================================

function checkEnemies() {

  for (
    let enemy of enemies
  ) {

    if (enemy.dead) {
      continue;
    }


    if (
      player.x <
      enemy.x + enemy.w &&
      player.x + player.w >
      enemy.x &&
      player.y <
      enemy.y + enemy.h &&
      player.y + player.h >
      enemy.y
    ) {


      // 踏む
      if (
        player.vy > 0 &&
        player.y + player.h <
        enemy.y + 18
      ) {

        enemy.dead = true;

        player.vy = -8;

        score += 200;

        continue;
      }


      // スター
      if (
        player.star > 0
      ) {

        enemy.dead = true;

        score += 200;

        continue;
      }


      damage();
    }
  }
}


// ==========================================
// ダメージ
// ==========================================

function damage() {

  if (
    player.invincible > 0
  ) {
    return;
  }


  lives--;


  if (lives <= 0) {

    gameState = "gameover";

    return;
  }


  player.x =
    max(
      50,
      player.x - 300
    );

  player.y = 250;

  player.vx = 0;
  player.vy = 0;

  player.invincible = 120;
}


// ==========================================
// 描画
// ==========================================

function drawWorld() {

  for (
    let p of platforms
  ) {

    p.draw();
  }


  for (
    let b of blocks
  ) {

    b.draw();
  }


  drawCoins();


  for (
    let item of items
  ) {

    item.draw();
  }


  for (
    let enemy of enemies
  ) {

    enemy.draw();
  }


  drawGoal();

  player.draw();
}


// ==========================================
// コイン描画
// ==========================================

function drawCoins() {

  for (
    let c of coins
  ) {

    if (c.taken) {
      continue;
    }


    c.angle += 0.1;


    let w =
      abs(
        sin(c.angle)
      ) * 15 + 5;


    fill("#ffd21f");

    ellipse(
      c.x - cameraX,

// ==========================================
// キーボード
// ==========================================

function keyPressed() {

  // ジャンプ
  if (
    keyCode === UP_ARROW ||
    key === " "
  ) {

    if (
      gameState === "play"
    ) {

      player.jump();
    }

    return false;
  }


  // 次のレベル
  if (
    keyCode === ENTER &&
    gameState === "levelClear"
  ) {

    if (level < 5) {

      startLevel(level + 1);

    } else {

      gameState = "allClear";
    }

    return false;
  }


  // リスタート
  if (
    key === "r" ||
    key === "R"
  ) {

    score = 0;

    lives = 3;

    startLevel(1);

    return false;
  }
}


// ==========================================
// マウス
// ==========================================

function mousePressed() {

  if (
    gameState === "play"
  ) {

    player.jump();

    return;
  }


  if (
    gameState === "levelClear"
  ) {

    if (level < 5) {

      startLevel(level + 1);

    } else {

      gameState = "allClear";
    }

    return;
  }


  if (
    gameState === "gameover" ||
    gameState === "allClear"
  ) {

    score = 0;

    lives = 3;

    startLevel(1);
  }
}
