// ハンターのマジック風 p5.js プロトタイプ
// ==========================================

let player;
let enemies = [];
let bullets = [];
let gems = [];

let gameState = "playing";

let level = 1;
let exp = 0;
let expToNext = 10;

let spawnTimer = 0;
let score = 0;

// 選択できるスキル
let skills = [
  {
    name: "火球強化",
    description: "魔法のダメージ +8",
    apply: () => {
      player.damage += 8;
    }
  },
  {
    name: "高速詠唱",
    description: "攻撃速度アップ",
    apply: () => {
      player.attackInterval *= 0.75;
    }
  },
  {
    name: "魔法弾追加",
    description: "同時に撃つ魔法が +1",
    apply: () => {
      player.multiShot++;
    }
  },
  {
    name: "最大HPアップ",
    description: "最大HP +30、HP全回復",
    apply: () => {
      player.maxHp += 30;
      player.hp = player.maxHp;
    }
  }
];

let skillChoices = [];

// ==========================================
// setup
// ==========================================

function setup() {
  createCanvas(800, 600);

  player = {
    x: width / 2,
    y: height / 2,

    radius: 22,

    hp: 100,
    maxHp: 100,

    damage: 15,

    attackInterval: 45,
    attackTimer: 0,

    multiShot: 1,

    speed: 3
  };
}

// ==========================================
// draw
// ==========================================

function draw() {
  background(18, 20, 32);

  if (gameState === "playing") {
    updateGame();
    drawGame();
    drawUI();
  }

  if (gameState === "levelup") {
    drawGame();
    drawUI();
    drawLevelUp();
  }

  if (gameState === "gameover") {
    drawGame();
    drawUI();
    drawGameOver();
  }
}

// ==========================================
// ゲーム更新
// ==========================================

function updateGame() {

  // 敵を一定時間ごとに出す
  spawnTimer++;

  if (spawnTimer > 45) {
    spawnEnemy();
    spawnTimer = 0;
  }

  // プレイヤーをマウスで移動
  player.x = mouseX;
  player.y = mouseY;

  player.x = constrain(player.x, 30, width - 30);
  player.y = constrain(player.y, 60, height - 30);

  // 自動攻撃
  player.attackTimer++;

  if (player.attackTimer >= player.attackInterval) {
    shootMagic();
    player.attackTimer = 0;
  }

  // 魔法弾更新
  for (let i = bullets.length - 1; i >= 0; i--) {

    let b = bullets[i];

    b.x += b.vx;
    b.y += b.vy;

    b.life--;

    // 敵との当たり判定
    for (let j = enemies.length - 1; j >= 0; j--) {

      let e = enemies[j];

      let d = dist(b.x, b.y, e.x, e.y);

      if (d < b.radius + e.radius) {

        e.hp -= b.damage;

        bullets.splice(i, 1);

        // 敵を倒した
        if (e.hp <= 0) {

          gems.push({
            x: e.x,
            y: e.y,
            value: 1
          });

          enemies.splice(j, 1);

          score += 10;
        }

        break;
      }
    }

    if (i < bullets.length && b.life <= 0) {
      bullets.splice(i, 1);
    }
  }

  // 経験値ジェム
  for (let i = gems.length - 1; i >= 0; i--) {

    let g = gems[i];

    let d = dist(player.x, player.y, g.x, g.y);

    // プレイヤーに近づくと吸い寄せる
    if (d < 120) {

      let angle = atan2(
        player.y - g.y,
        player.x - g.x
      );

      g.x += cos(angle) * 5;
      g.y += sin(angle) * 5;
    }

    if (d < player.radius + 10) {

      exp += g.value;

      gems.splice(i, 1);

      checkLevelUp();
    }
  }

  // 敵をプレイヤーへ移動
  for (let e of enemies) {

    let angle = atan2(
      player.y - e.y,
      player.x - e.x
    );

    e.x += cos(angle) * e.speed;
    e.y += sin(angle) * e.speed;

    // プレイヤーに接触
    let d = dist(player.x, player.y, e.x, e.y);

    if (d < player.radius + e.radius) {

      player.hp -= 0.5;

      if (player.hp <= 0) {
        player.hp = 0;
        gameState = "gameover";
      }
    }
  }
}

// ==========================================
// 敵を生成
// ==========================================

function spawnEnemy() {

  let side = floor(random(4));

  let x;
  let y;

  if (side === 0) {
    x = random(width);
    y = -30;
  }

  if (side === 1) {
    x = width + 30;
    y = random(height);
  }

  if (side === 2) {
    x = random(width);
    y = height + 30;
  }

  if (side === 3) {
    x = -30;
    y = random(height);
  }

  enemies.push({
    x: x,
    y: y,

    radius: 18,

    hp: 30,
    maxHp: 30,

    speed: random(0.7, 1.4)
  });
}

// ==========================================
// 魔法を撃つ
// ==========================================

function shootMagic() {

  if (enemies.length === 0) return;

  // 一番近い敵を探す
  let targets = [...enemies];

  targets.sort((a, b) => {

    let da = dist(player.x, player.y, a.x, a.y);
    let db = dist(player.x, player.y, b.x, b.y);

    return da - db;
  });

  let count = min(player.multiShot, targets.length);

  for (let i = 0; i < count; i++) {

    let target = targets[i];

    let angle = atan2(
      target.y - player.y,
      target.x - player.x
    );

    // 複数弾なら少し角度をずらす
    if (count > 1) {
      angle += map(
        i,
        0,
        count - 1,
        -0.25,
        0.25
      );
    }

    bullets.push({
      x: player.x,
      y: player.y,

      vx: cos(angle) * 8,
      vy: sin(angle) * 8,

      radius: 8,

      damage: player.damage,

      life: 100
    });
  }
}

// ==========================================
// レベルアップ判定
// ==========================================

function checkLevelUp() {

  if (exp >= expToNext) {

    exp -= expToNext;

    level++;

    expToNext = floor(expToNext * 1.35);

    showLevelUp();
  }
}

// ==========================================
// レベルアップ画面
// ==========================================

function showLevelUp() {

  gameState = "levelup";

  skillChoices = [];

  // ランダムに3つ選ぶ
  let pool = [...skills];

  for (let i = 0; i < 3; i++) {

    let index = floor(random(pool.length));

    skillChoices.push(pool[index]);

    pool.splice(index, 1);
  }
}

// ==========================================
// 描画
// ==========================================

function drawGame() {

  // 背景のグリッド
  stroke(35, 38, 55);

  for (let x = 0; x < width; x += 40) {
    line(x, 0, x, height);
  }

  for (let y = 0; y < height; y += 40) {
    line(0, y, width, y);
  }

  noStroke();

  // 経験値ジェム
  for (let g of gems) {

    fill(80, 220, 255);

    circle(g.x, g.y, 10);

    fill(180, 250, 255);

    circle(g.x - 2, g.y - 2, 3);
  }

  // 敵
  for (let e of enemies) {

    fill(210, 60, 70);

    circle(e.x, e.y, e.radius * 2);

    // 敵HPバー
    fill(50);

    rect(
      e.x - 20,
      e.y - 28,
      40,
      5
    );

    fill(80, 230, 100);

    rect(
      e.x - 20,
      e.y - 28,
      40 * (e.hp / e.maxHp),
      5
    );
  }

  // 魔法弾
  for (let b of bullets) {

    fill(255, 180, 40);

    circle(b.x, b.y, b.radius * 2);

    fill(255, 240, 120);

    circle(b.x, b.y, 7);
  }

  // プレイヤー
  drawPlayer();
}

// ==========================================
// プレイヤー
// ==========================================

function drawPlayer() {

  // 魔法陣
  noFill();

  stroke(80, 180, 255);
  strokeWeight(2);

  circle(
    player.x,
    player.y,
    55
  );

  circle(
    player.x,
    player.y,
    45
  );

  // 本体
  noStroke();

  fill(80, 130, 255);

  circle(
    player.x,
    player.y,
    player.radius * 2
  );

  fill(230);

  circle(
    player.x,
    player.y - 5,
    10
  );
}

// ==========================================
// クリック
// ==========================================

function mousePressed() {

  if (gameState === "levelup") {

    for (let i = 0; i < 3; i++) {

      let x = 130 + i * 270;

      if (
        mouseX > x - 100 &&
        mouseX < x + 100 &&
        mouseY > 190 &&
        mouseY < 410
      ) {

        skillChoices[i].apply();

        gameState = "playing";

        break;
      }
    }
  }

  // ゲームオーバーならクリックでリスタート
  else if (gameState === "gameover") {

    restartGame();
  }
}

// ==========================================
// ゲームオーバー
// ==========================================

function drawGameOver() {

  fill(0, 0, 0, 190);

  rect(0, 0, width, height);

  textAlign(CENTER);

  fill(255, 80, 80);

  textSize(48);

  text(
    "GAME OVER",
    width / 2,
    250
  );

  fill(255);

  textSize(20);

  text(
    "Score: " + score,
    width / 2,
    300
  );

  text(
    "クリックでリスタート",
    width / 2,
    350
  );

  textAlign(LEFT);
}

// ==========================================
// リスタート
// ==========================================

function restartGame() {

  enemies = [];
  bullets = [];
  gems = [];

  level = 1;
  exp = 0;
  expToNext = 10;

  score = 0;

  spawnTimer = 0;

  player.hp = 100;
  player.maxHp = 100;

  player.damage = 15;

  player.attackInterval = 45;

  player.multiShot = 1;

  gameState = "playing";
}
