// ============================================================
// 3D FPS MAZE

// ============================================================
// MAZE
// ============================================================

const maze = [
  "###############",
  "#P            #",
  "# ##### ##### #",
  "#     #       #",
  "##### # #######",
  "#     #       #",
  "# ### ##### # #",
  "# #         # #",
  "# # ####### # #",
  "# #       #   #",
  "# ####### #####",
  "#             #",
  "###############"
];


// ============================================================
// SETTINGS
// ============================================================

const CELL_SIZE = 80;
const WALL_HEIGHT = 120;

const PLAYER_HEIGHT = 45;
const PLAYER_RADIUS = 18;

const MOVE_SPEED = 3.5;

const MOUSE_SENSITIVITY = 0.0025;


// ============================================================
// PLAYER
// ============================================================

let player = {
  x: 0,
  z: 0,
  angle: 0
};


// ============================================================
// WALLS
// ============================================================

let walls = [];


// ============================================================
// SETUP
// ============================================================

function setup() {

  createCanvas(
    windowWidth,
    windowHeight,
    WEBGL
  );

  createMaze();

  noStroke();

}


// ============================================================
// CREATE MAZE
// ============================================================

function createMaze() {

  const mazeWidth =
    maze[0].length;

  const mazeHeight =
    maze.length;


  const offsetX =
    mazeWidth * CELL_SIZE / 2;

  const offsetZ =
    mazeHeight * CELL_SIZE / 2;


  for (
    let row = 0;
    row < mazeHeight;
    row++
  ) {

    for (
      let col = 0;
      col < maze[row].length;
      col++
    ) {

      const tile =
        maze[row][col];


      const worldX =
        col * CELL_SIZE
        - offsetX
        + CELL_SIZE / 2;


      const worldZ =
        row * CELL_SIZE
        - offsetZ
        + CELL_SIZE / 2;


      if (tile === "#") {

        walls.push({

          x: worldX,
          z: worldZ

        });

      }


      if (tile === "P") {

        player.x = worldX;
        player.z = worldZ;

      }

    }

  }

}


// ============================================================
// DRAW
// ============================================================

function draw() {

  background(
    35,
    40,
    50
  );


  // --------------------------
  // Player
  // --------------------------

  updatePlayer();


  // --------------------------
  // Camera
  // --------------------------

  updateCamera();


  // --------------------------
  // Lighting
  // --------------------------

  ambientLight(
    150
  );


  directionalLight(

    255,
    255,
    255,

    -0.3,
    -1,
    -0.5

  );


  // --------------------------
  // World
  // --------------------------

  drawFloor();

  drawWalls();

}


// ============================================================
// PLAYER
// ============================================================

function updatePlayer() {

  let forward = 0;
  let strafe = 0;


  // ------------------------------------------------
  // Read keyboard EVERY frame
  // ------------------------------------------------

  if (
    keyIsDown(87) &&
    document.activeElement !== document.body
  ) {
    forward += 1;
  }

  if (
    keyIsDown(83) &&
    document.activeElement !== document.body
  ) {
    forward -= 1;
  }

  if (
    keyIsDown(65) &&
    document.activeElement !== document.body
  ) {
    strafe -= 1;
  }

  if (
    keyIsDown(68) &&
    document.activeElement !== document.body
  ) {
    strafe += 1;
  }


  // ------------------------------------------------
  // No movement
  // ------------------------------------------------

  if (
    forward === 0 &&
    strafe === 0
  ) {
    return;
  }


  // ------------------------------------------------
  // Forward direction
  // ------------------------------------------------

  let dx =
    sin(player.angle) *
    forward;


  let dz =
    -cos(player.angle) *
    forward;


  // ------------------------------------------------
  // Strafe
  // ------------------------------------------------

  dx +=
    cos(player.angle) *
    strafe;


  dz +=
    sin(player.angle) *
    strafe;


  // ------------------------------------------------
  // Normalize
  // ------------------------------------------------

  const length =
    sqrt(
      dx * dx +
      dz * dz
    );


  if (length === 0) {
    return;
  }


  dx /= length;
  dz /= length;


  // ------------------------------------------------
  // Speed
  // ------------------------------------------------

  dx *= MOVE_SPEED;
  dz *= MOVE_SPEED;


  // ------------------------------------------------
  // Collision
  // ------------------------------------------------

  moveWithCollision(
    dx,
    dz
  );

}


// ============================================================
// MOVE WITH COLLISION
// ============================================================

function moveWithCollision(
  dx,
  dz
) {

  const nextX =
    player.x + dx;


  if (
    !isColliding(
      nextX,
      player.z
    )
  ) {

    player.x = nextX;

  }


  const nextZ =
    player.z + dz;


  if (
    !isColliding(
      player.x,
      nextZ
    )
  ) {

    player.z = nextZ;

  }

}


// ============================================================
// COLLISION
// ============================================================

function isColliding(
  x,
  z
) {

  for (
    const wall of walls
  ) {

    const closestX =
      constrain(

        x,

        wall.x - CELL_SIZE / 2,
        wall.x + CELL_SIZE / 2

      );


    const closestZ =
      constrain(

        z,

        wall.z - CELL_SIZE / 2,
        wall.z + CELL_SIZE / 2

      );


    const distance =
      dist(

        x,
        z,

        closestX,
        closestZ

      );


    if (
      distance < PLAYER_RADIUS
    ) {

      return true;

    }

  }


  return false;

}


// ============================================================
// CAMERA
// ============================================================

function updateCamera() {

  const lookDistance = 100;


  const lookX =
    player.x +
    sin(player.angle) *
    lookDistance;


  const lookZ =
    player.z -
    cos(player.angle) *
    lookDistance;


  camera(

    player.x,
    -PLAYER_HEIGHT,
    player.z,

    lookX,
    -PLAYER_HEIGHT,
    lookZ,

    0,
    1,
    0

  );

}


// ============================================================
// FLOOR
// ============================================================

function drawFloor() {

  push();


  translate(
    0,
    20,
    0
  );


  rotateX(
    HALF_PI
  );


  // Base color

  ambientMaterial(
    70,
    75,
    80
  );


  plane(
    3000,
    3000
  );


  pop();


  // Floor grid

  drawFloorGrid();

}


// ============================================================
// FLOOR GRID
// ============================================================

function drawFloorGrid() {

  push();


  translate(
    0,
    19,
    0
  );


  rotateX(
    HALF_PI
  );


  stroke(
    110,
    115,
    120
  );

  strokeWeight(1);


  const size = 1500;

  const step = CELL_SIZE;


  for (
    let x = -size;
    x <= size;
    x += step
  ) {

    line(
      x,
      -size,
      x,
      size
    );

  }


  for (
    let z = -size;
    z <= size;
    z += step
  ) {

    line(
      -size,
      z,
      size,
      z
    );

  }


  noStroke();


  pop();

}


// ============================================================
// WALLS
// ============================================================

function drawWalls() {

  for (
    const wall of walls
  ) {

    drawWall(
      wall.x,
      wall.z
    );

  }

}


// ============================================================
// SINGLE WALL
// ============================================================

function drawWall(
  x,
  z
) {

  push();


  translate(
    x,
    -30,
    z
  );


  // Base wall

  ambientMaterial(
    145,
    155,
    170
  );


  box(
    CELL_SIZE,
    WALL_HEIGHT,
    CELL_SIZE
  );


  pop();


  // Brick pattern

  drawBrickPattern(
    x,
    z
  );

}


// ============================================================
// BRICK PATTERN
// ============================================================

function drawBrickPattern(
  x,
  z
) {

  const rows = 5;

  const brickHeight =
    WALL_HEIGHT / rows;


  for (
    let row = 0;
    row < rows;
    row++
  ) {

    const y =
      -WALL_HEIGHT / 2
      + brickHeight / 2
      + row * brickHeight;


    const offset =
      row % 2 === 0
        ? 0
        : CELL_SIZE / 2;


    // Front

    drawBrickFace(
      x,
      y - 30,
      z - CELL_SIZE / 2 - 0.5,
      0,
      offset,
      brickHeight
    );


    // Back

    drawBrickFace(
      x,
      y - 30,
      z + CELL_SIZE / 2 + 0.5,
      PI,
      offset,
      brickHeight
    );

  }

}


// ============================================================
// BRICK FACE
// ============================================================

function drawBrickFace(
  x,
  y,
  z,
  rotation,
  offset,
  brickHeight
) {

  push();


  translate(
    x,
    y,
    z
  );


  rotateY(
    rotation
  );


  // Horizontal lines

  stroke(
    80,
    85,
    95
  );

  strokeWeight(2);


  line(
    -CELL_SIZE / 2,
    -brickHeight / 2,
    0,

    CELL_SIZE / 2,
    -brickHeight / 2,
    0
  );


  line(
    -CELL_SIZE / 2,
    brickHeight / 2,
    0,

    CELL_SIZE / 2,
    brickHeight / 2,
    0
  );


  // Vertical brick lines

  const brickWidth = 40;


  for (
    let i = -2;
    i <= 2;
    i++
  ) {

    const bx =
      i * brickWidth
      + offset % brickWidth;


    line(
      bx,
      -brickHeight / 2,
      0,

      bx,
      brickHeight / 2,
      0
    );

  }


  noStroke();


  pop();

}


// ============================================================
// MOUSE
// ============================================================

function mousePressed() {

  if (
    document.pointerLockElement
  ) {

    return;

  }


  requestPointerLock();

}


// ============================================================
// MOUSE LOOK
// ============================================================

function mouseMoved() {

  if (
    document.pointerLockElement
  ) {

    player.angle +=
      movedX *
      MOUSE_SENSITIVITY;

  }

}


// ============================================================
// KEY RESET
// ============================================================

function keyReleased() {

  // Force browser to release
  // any possible stuck key.

  if (
    key === "w" ||
    key === "W"
  ) {
    return;
  }

  if (
    key === "a" ||
    key === "A"
  ) {
    return;
  }

  if (
    key === "s" ||
    key === "S"
  ) {
    return;
  }

  if (
    key === "d" ||
    key === "D"
  ) {
    return;
  }

}

  function() {

    // Nothing is stored as movement state.

  }
);

function windowResized() {

  resizeCanvas(

  );

}
